===== CSV OUTPUT IMPLEMENTATION =====
import Papa from "papaparse";
import type {
  CompanyResearchReport,
  ResearchOutputRow,
} from "./runner";
export const RESEARCH_COLUMNS = [
  "company_name",
  "website",
  "claim_type",
  "claim",
  "source_url",
  "source_title",
  "retrieved_at",
  "evidence",
] as const;
export function createResearchCsv(rows: ResearchOutputRow[]): string {
  return Papa.unparse({
    fields: [...RESEARCH_COLUMNS],
    data: rows.map((row) =>
      RESEARCH_COLUMNS.map((column) => row[column]),
    ),
  });
}
export function createResearchReport(
  companies: CompanyResearchReport[],
): string {
  const lines = [
    "# Ledgerwise company research report",
    "",
    Companies processed: ${companies.length},
    Pages read: ${companies.reduce((sum, company) => sum + company.pagesRead.length, 0)},
    Claims proposed: ${companies.reduce((sum, company) => sum + company.proposed, 0)},
    Claims accepted: ${companies.reduce((sum, company) => sum + company.accepted, 0)},
    Claims rejected: ${companies.reduce((sum, company) => sum + company.rejected, 0)},
    "",
    "> Accepted means the evidence quotation was found in the fetched page text. It does not independently prove that the claim is true. Human review is required before importing.",
    "",
  ];
  for (const company of companies) {
    lines.push(
      ## ${company.companyName},
      "",
      - Website: ${company.website},
      - Pages read: ${company.pagesRead.length},
      - Claims proposed: ${company.proposed},
      - Claims accepted: ${company.accepted},
      - Claims rejected: ${company.rejected},
      "",
    );
if (company.pagesRead.length) {
  lines.push("### Pages read", "");
  for (const url of company.pagesRead) {
    lines.push(`- ${url}`);
  }
  lines.push("");
}

if (company.rejectedClaims.length) {
  lines.push("### Rejected claims", "");
  for (const rejected of company.rejectedClaims) {
    lines.push(`- **${rejected.claim_type || "Unknown type"}:** ${rejected.reason}`);
  }
  lines.push("");
}

if (company.errors.length) {
  lines.push("### Errors", "");
  for (const error of company.errors) {
    lines.push(`- ${error}`);
  }
  lines.push("");
}

  }
  return ${lines.join("\n").trim()}\n;
}

test("CSV output uses the exact importer columns and preserves values", () => {
  const rows: ResearchOutputRow[] = [
    {
      company_name: "Acme Goods",
      website: "https://example.com/",
      claim_type: "category",
      claim: "Sells home goods",
      source_url: "https://example.com/about",
      source_title: "About Acme",
      retrieved_at: "2026-11-13",
      evidence: 'We sell "home goods" online.',
    },
  ];
  const csv = createResearchCsv(rows);
  const parsed = Papa.parse<Record<string, string>>(csv, {
    header: true,
    skipEmptyLines: true,
  });
  assert.deepEqual(parsed.meta.fields, [...RESEARCH_COLUMNS]);
  assert.equal(parsed.data.length, 1);
  assert.equal(parsed.data[0].company_name, "Acme Goods");
  assert.equal(parsed.data[0].evidence, 'We sell "home goods" online.');
});

const MAX_PAGE_CHARACTERS = 120_000;
function safeHttpUrl(input: string): URL {
  const url = new URL(input);
  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error("Only HTTP and HTTPS pages are supported");
  }
  if (url.username || url.password) {
    throw new Error("URLs containing credentials are not allowed");
  }
  if (
    url.hostname === "localhost" ||
    url.hostname.endsWith(".localhost") ||
    url.hostname.endsWith(".local") ||
    url.hostname === "127.0.0.1" ||
    url.hostname === "0.0.0.0" ||
    url.hostname.startsWith("10.") ||
    url.hostname.startsWith("192.168.") ||
    /^172.(1[6-9]|2\d|3[01])./.test(url.hostname)
  ) {
    throw new Error("Private or local URLs are not allowed");
  }
  url.hash = "";
  return url;
}
function fallbackTitle(url: URL): string {
  const path = decodeURIComponent(url.pathname)
    .split("/")
    .filter(Boolean)
    .pop();
  return path
    ? path.replace(/[-_]/g, " ").replace(/.[^.]+$/, "")
    : url.hostname;
}
export function parseReaderResponse(
  responseText: string,
  originalUrl: string,
): ResearchPage {
  const url = safeHttpUrl(originalUrl);
  const lines = responseText.replace(/\r\n/g, "\n").split("\n");
  const titleLine = lines.find((line) => /^Title:\s*/i.test(line));
  const title = titleLine?.replace(/^Title:\s*/i, "").trim();
  const markerIndex = lines.findIndex((line) =>
    /^Markdown Content:\s*$/i.test(line.trim()),
  );
  let contentLines =
    markerIndex >= 0 ? lines.slice(markerIndex + 1) : lines;
  // Remove Jina's metadata headers when the Markdown Content marker is absent.
  if (markerIndex < 0) {
    contentLines = contentLines.filter(
      (line) =>
        !/^(Title|URL Source|Markdown Content):\s*/i.test(line.trim()),
    );
  }
  const text = contentLines.join("\n").trim().slice(0, MAX_PAGE_CHARACTERS);
  if (!text) {
    throw new Error(Reader returned no page text for ${url.toString()});
  }
  return {
    url: url.toString(),
    title: title || fallbackTitle(url),
    text,
  };
}
const USEFUL_PATH = /about|company|our-story|team|leadership|contact|career|jobs|press|news|shipping|wholesale|retail|stockist/i;
export function discoverSameSiteLinks(
  markdown: string,
  baseUrl: string,
  maxLinks = 3,
): string[] {
  const base = safeHttpUrl(baseUrl);
  const found = new Map<string, { url: string; score: number }>();
  const markdownLink = /](([^)\s]+)(?:\s+"[^"]*")?)/g;
  let match: RegExpExecArray | null;
  while ((match = markdownLink.exec(markdown)) !== null) {
    const raw = match[1];
try {
  const candidate = new URL(raw, base);

  if (!["http:", "https:"].includes(candidate.protocol)) continue;
  if (candidate.hostname !== base.hostname) continue;
  if (candidate.username || candidate.password) continue;

  candidate.hash = "";
  candidate.search = "";

  // Skip the current page and common non-page assets.
  if (candidate.pathname === base.pathname) continue;
  if (/\.(pdf|jpg|jpeg|png|gif|webp|svg|css|js|zip|xml)$/i.test(candidate.pathname)) {
    continue;
  }

  const normalized = candidate.toString();
  const score = USEFUL_PATH.test(candidate.pathname) ? 1 : 0;

  if (!found.has(normalized)) {
    found.set(normalized, { url: normalized, score });
  }
} catch {
  // Ignore malformed links; one bad link should not stop page discovery.
}

  }
  return [...found.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.max(0, maxLinks))
    .map((item) => item.url);
}

function splitText(text: string): string[] {
  if (text.length <= CHUNK_SIZE) return [text];
  const chunks: string[] = [];
  let start = 0;
  while (start < text.length) {
    const end = Math.min(start + CHUNK_SIZE, text.length);
    chunks.push(text.slice(start, end));
if (end === text.length) break;
start = end - CHUNK_OVERLAP;

  }
  return chunks;
}

) {

export function validateResearchUrl(input: string): URL {
  let url: URL;
  try {
    url = new URL(input);
  } catch {
    throw new Error("Invalid research URL");
  }
  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error("Only HTTP and HTTPS pages are supported");
  }
  if (url.username || url.password) {
    throw new Error("URLs containing credentials are not allowed");
  }
  const hostname = url.hostname.toLowerCase().replace(/\.$/, "");
  const ipHostname = hostname.replace(/^\[|\]$/g, "");
  // Research targets must use public DNS hostnames, not IP literals.
  // This avoids IPv4, IPv6, mapped IPv4, loopback and private IP variants.
  if (isIP(ipHostname) !== 0) {
    throw new Error("IP-address URLs are not allowed");
  }
  if (
    hostname === "localhost" ||
    hostname.endsWith(".localhost") ||
    hostname.endsWith(".local") ||
    hostname === "local"
  ) {
    throw new Error("Private or local URLs are not allowed");
  }
  url.hash = "";
  return url;
}

export function parseReaderResponse(
  responseText: string,
  originalUrl: string,
): ResearchPage {
  const url = safeHttpUrl(originalUrl);

  const titleLine = lines.find((line) => /^Title:\s*/i.test(line));

  const markerIndex = lines.findIndex((line) =>

  );
  let contentLines =
    markerIndex >= 0 ? lines.slice(markerIndex + 1) : lines;
  // Remove Jina's metadata headers when the Markdown Content marker is absent.
  if (markerIndex < 0) {
    contentLines = contentLines.filter(

        !/^(Title|URL Source|Markdown Content):\s*/i.test(line.trim()),
    );
  }
  const text = contentLines.join("\n").trim().slice(0, MAX_PAGE_CHARACTERS);
  if (!text) {
    throw new Error(Reader returned no page text for ${url.toString()});
  }
  return {
    url: url.toString(),
    title: title || fallbackTitle(url),
    text,
  };

export function validateResearchUrl(input: string): URL {
  let url: URL;
  try {
    url = new URL(input);
  } catch {
    throw new Error("invalid URL");
  }
  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error("Only HTTP and HTTPS pages are supported");
  }
  if (url.username || url.password) {
    throw new Error("URLs containing credentials are not allowed");
  }
  const hostname = url.hostname.toLowerCase().replace(/.$/, "");
  const ipHostname =
    hostname.startsWith("[") && hostname.endsWith("]")
      ? hostname.slice(1, -1)
      : hostname;
  if (isIP(ipHostname) !== 0) {
    throw new Error("Private or local IP-address URLs are not allowed");
  }
  if (
    hostname === "localhost" ||
    hostname.endsWith(".localhost") ||
    hostname.endsWith(".local") ||
    hostname === "local"
  ) {
    throw new Error("Private or local URLs are not allowed");
  }
  url.hash = "";
  return url;
}

    );
  }
  const text = contentLines.join("\n").trim().slice(0, MAX_PAGE_CHARACTERS);
  if (!text) {
    throw new Error(Reader returned no page text for ${url.toString()});
  }
  return {
    url: url.toString(),
    title: title || fallbackTitle(url),
    text,
  };
}
const USEFUL_PATH = /about|company|our-story|team|leadership|contact|career|jobs|press|news|shipping|wholesale|retail|stockist/i;
export function discoverSameSiteLinks(
  markdown: string,
  baseUrl: string,
  maxLinks = 3,
): string[] {
  const base = safeHttpUrl(baseUrl);
  const found = new Map<string, { url: string; score: number }>();
  const markdownLink = /](([^)\s]+)(?:\s+"[^"]*")?)/g;
  let match: RegExpExecArray | null;
  while ((match = markdownLink.exec(markdown)) !== null) {
    const raw = match[1];
try {
  const candidate = new URL(raw, base);

  if (!["http:", "https:"].includes(candidate.protocol)) continue;
  if (candidate.hostname !== base.hostname) continue;
  if (candidate.username || candidate.password) continue;

  candidate.hash = "";
  candidate.search = "";

  // Skip the current page and common non-page assets.
  if (candidate.pathname === base.pathname) continue;
  if (/\.(pdf|jpg|jpeg|png|gif|webp|svg|css|js|zip|xml)$/i.test(candidate.pathname)) {
    continue;
  }

  const normalized = candidate.toString();
  const score = USEFUL_PATH.test(candidate.pathname) ? 1 : 0;

  if (!found.has(normalized)) {
    found.set(normalized, { url: normalized, score });
  }
} catch {
  // Ignore malformed links; one bad link should not stop page discovery.
}

  }
  return [...found.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.max(0, maxLinks))
    .map((item) => item.url);
}
export async function fetchResearchPage(
  originalUrl: string,
  deps: PageDeps = {},
): Promise {
  const target = safeHttpUrl(originalUrl);
  const fetcher = deps.fetch ?? fetch;
  const readerUrl = https://r.jina.ai/${target.toString()};
  const response = await fetcher(readerUrl, {
    method: "GET",
    headers: {
      Accept: "text/plain",
      "User-Agent": "LedgerwiseResearchPrototype/1.0",
    },
    signal: AbortSignal.timeout(30_000),
  });
  if (!response.ok) {
    throw new Error(
      Reader request failed for ${target.toString()}: HTTP ${response.status},
    );
  }
  const body = await response.text();
  return parseReaderResponse(body, target.toString());
}
