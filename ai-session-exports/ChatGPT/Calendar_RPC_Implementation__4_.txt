import { CRITERIA, type Criterion } from "./config";
export type SourceRow = {
  id: string;
  claim: string;
  claim_type: string;
  source_url: string;
  source_title: string;
  retrieved_at: string;
  evidence: string;
  verification_status?: string;
  reviewed_by?: string | null;
  reviewed_at?: string | null;
};
export type CriterionResult = {
  criterion: Criterion;
  met: boolean;
  points: number | null;
  reason: string;
  sources: SourceRow[];
};
export type ScoreResult = {
  criteria: CriterionResult[];
  disqualifiers: SourceRow[];
  weightsComplete: boolean;
  total: number | null;
  maxTotal: number | null;
  provisional: boolean;
};
function isHumanVerified(source: SourceRow): boolean {
  return (
    source.verification_status === "verified" &&
    Boolean(source.reviewed_by) &&
    Boolean(source.reviewed_at)
  );
}
export function scoreCompany(
  sources: SourceRow[],
  criteria: Criterion[] = CRITERIA
): ScoreResult {
  const weightsComplete = criteria.every(
    (criterion) =>
      criterion.weight !== null &&
      Number.isInteger(criterion.weight) &&
      criterion.weight > 0
  );
  const verifiedSources = sources.filter(isHumanVerified);
  const results: CriterionResult[] = criteria.map((criterion) => {
    const matched = verifiedSources.filter(
      (source) => source.claim_type === criterion.claimType
    );
    const met = matched.length > 0;
    const points =
      criterion.weight === null ? null : met ? criterion.weight : 0;
const hasUnverifiedEvidence = sources.some(
  (source) => source.claim_type === criterion.claimType
);

const reason = met
  ? matched.map((source) => source.claim).join("; ")
  : hasUnverifiedEvidence
    ? "Evidence exists but has not been human-verified; this criterion remains unknown."
    : "No human-verified evidence recorded; this criterion remains unknown.";

return {
  criterion,
  met,
  points,
  reason,
  sources: matched,
};

  });
  const disqualifiers = verifiedSources.filter(
    (source) => source.claim_type === "disqualifier"
  );
  const provisional = results.some((result) => !result.met);
  return {
    criteria: results,
    disqualifiers,
    weightsComplete,
    total: weightsComplete
      ? results.reduce((sum, result) => sum + (result.points ?? 0), 0)
      : null,
    maxTotal: weightsComplete
      ? criteria.reduce((sum, criterion) => sum + (criterion.weight ?? 0), 0)
      : null,
    provisional,
  };
}
''')

"use server";
import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";
import { getSessionUser } from "@/lib/server/auth";
import { writeAudit } from "@/lib/server/audit";
import { CRITERIA } from "@/lib/scoring/config";
import { scoreCompany, type SourceRow } from "@/lib/scoring/score";
export type GenerateScoresResult = {
  ok: boolean;
  companiesScored?: number;
  scoreRowsCreated?: number;
  warning?: string;
  error?: string;
};
type CompanyRow = { id: string; name: string };
export async function generateQualificationScores(): Promise {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);
  if (!user) return { ok: false, error: "Not signed in." };
  const totalWeight = CRITERIA.reduce(
    (sum, criterion) => sum + (criterion.weight ?? 0),
    0
  );
  if (
    CRITERIA.some(
      (criterion) =>
        criterion.weight === null ||
        !Number.isInteger(criterion.weight) ||
        criterion.weight <= 0
    ) ||
    totalWeight !== 100
  ) {
    return {
      ok: false,
      error: "Scoring configuration is invalid. Criteria weights must total 100.",
    };
  }
  const [
    { data: companies, error: companyError },
    { data: sources, error: sourceError },
  ] = await Promise.all([
    supabase.from("companies").select("id,name").order("name").limit(5000),
    supabase
      .from("company_sources")
      .select(
        "id,company_id,claim,claim_type,source_url,source_title,retrieved_at,evidence,verification_status,reviewed_by,reviewed_at"
      )
      .limit(10000),
  ]);
  if (companyError) return { ok: false, error: companyError.message };
  if (sourceError) return { ok: false, error: sourceError.message };
  const companyRows = (companies ?? []) as CompanyRow[];
  if (companyRows.length === 0) {
    return { ok: false, error: "There are no companies to score." };
  }
  const sourcesByCompany = new Map<string, SourceRow[]>();
  for (const source of (sources ?? []) as (SourceRow & {
    company_id: string;
  })[]) {
    const current = sourcesByCompany.get(source.company_id) ?? [];
    current.push(source);
    sourcesByCompany.set(source.company_id, current);
  }
  const batchTimestamp = new Date().toISOString();
  const scoreRows = companyRows.flatMap((company) => {
    const score = scoreCompany(sourcesByCompany.get(company.id) ?? []);
    return score.criteria.map((result) => ({
      company_id: company.id,
      criterion: result.criterion.key,
      weight: result.criterion.weight!,
      points: result.points ?? 0,
      reason: result.reason,
      source_id:
        result.points && result.sources.length > 0
          ? result.sources[0].id
          : null,
      created_at: batchTimestamp,
    }));
  });
  const { error: insertError } = await supabase
    .from("qualification_scores")
    .insert(scoreRows);
  if (insertError) return { ok: false, error: insertError.message };
  const audit = await writeAudit(supabase, {
    actorId: user.id,
    action: "qualification_scores_generated",
    entityType: "qualification_scores",
    detail: {
      companies_scored: companyRows.length,
      score_rows_created: scoreRows.length,
      scoring_batch: batchTimestamp,
      weights: Object.fromEntries(
        CRITERIA.map((criterion) => [criterion.key, criterion.weight])
      ),
      evidence_rule: "Only human-verified sources earn positive points.",
    },
  });
  revalidatePath("/prospects");
  for (const company of companyRows) {
    revalidatePath(/prospects/${company.id});
  }
  return {
    ok: true,
    companiesScored: companyRows.length,
    scoreRowsCreated: scoreRows.length,
    warning: audit.ok
      ? undefined
      : Scores were saved, but the audit entry failed: ${audit.error},
  };
}
''')

import { CRITERIA, type Criterion } from "./config";
export type SourceRow = {
  id: string;
  claim: string;
  claim_type: string;
  source_url: string;
  source_title: string;
  retrieved_at: string;
  evidence: string;
  verification_status?: string;
  reviewed_by?: string | null;
  reviewed_at?: string | null;
};
export type CriterionResult = {
  criterion: Criterion;
  met: boolean;
  points: number | null;
  reason: string;
  sources: SourceRow[];
};
export type ScoreResult = {
  criteria: CriterionResult[];
  disqualifiers: SourceRow[];
  weightsComplete: boolean;
  total: number | null;
  maxTotal: number | null;
  provisional: boolean;
};
function isHumanVerified(source: SourceRow): boolean {
  return (
    source.verification_status === "verified" &&
    Boolean(source.reviewed_by) &&
    Boolean(source.reviewed_at)
  );
}
export function scoreCompany(
  sources: SourceRow[],
  criteria: Criterion[] = CRITERIA
): ScoreResult {
  const weightsComplete = criteria.every(
    (criterion) =>
      criterion.weight !== null &&
      Number.isInteger(criterion.weight) &&
      criterion.weight > 0
  );
  const verifiedSources = sources.filter(isHumanVerified);
  const results: CriterionResult[] = criteria.map((criterion) => {
    const matched = verifiedSources.filter(
      (source) => source.claim_type === criterion.claimType
    );
    const met = matched.length > 0;
    const points =
      criterion.weight === null ? null : met ? criterion.weight : 0;
const hasUnverifiedEvidence = sources.some(
  (source) => source.claim_type === criterion.claimType
);

const reason = met
  ? matched.map((source) => source.claim).join("; ")
  : hasUnverifiedEvidence
    ? "Evidence exists but has not been human-verified; this criterion remains unknown."
    : "No human-verified evidence recorded; this criterion remains unknown.";

return {
  criterion,
  met,
  points,
  reason,
  sources: matched,
};

  });
  const disqualifiers = verifiedSources.filter(
    (source) => source.claim_type === "disqualifier"
  );
  const provisional = results.some((result) => !result.met);
  return {
    criteria: results,
    disqualifiers,
    weightsComplete,
    total: weightsComplete
      ? results.reduce((sum, result) => sum + (result.points ?? 0), 0)
      : null,
    maxTotal: weightsComplete
      ? criteria.reduce((sum, criterion) => sum + (criterion.weight ?? 0), 0)
      : null,
    provisional,
  };
}
''')

"use server";
import { revalidatePath } from "next/cache";
import { createClient } from "@/lib/supabase/server";
import { getSessionUser } from "@/lib/server/auth";
import { writeAudit } from "@/lib/server/audit";
import { CRITERIA } from "@/lib/scoring/config";
import { scoreCompany, type SourceRow } from "@/lib/scoring/score";
export type GenerateScoresResult = {
  ok: boolean;
  companiesScored?: number;
  scoreRowsCreated?: number;
  warning?: string;
  error?: string;
};
type CompanyRow = { id: string; name: string };
export async function generateQualificationScores(): Promise {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);
  if (!user) return { ok: false, error: "Not signed in." };
  const totalWeight = CRITERIA.reduce(
    (sum, criterion) => sum + (criterion.weight ?? 0),
    0
  );
  if (
    CRITERIA.some(
      (criterion) =>
        criterion.weight === null ||
        !Number.isInteger(criterion.weight) ||
        criterion.weight <= 0
    ) ||
    totalWeight !== 100
  ) {
    return {
      ok: false,
      error: "Scoring configuration is invalid. Criteria weights must total 100.",
    };
  }
  const [
    { data: companies, error: companyError },
    { data: sources, error: sourceError },
  ] = await Promise.all([
    supabase.from("companies").select("id,name").order("name").limit(5000),
    supabase
      .from("company_sources")
      .select(
        "id,company_id,claim,claim_type,source_url,source_title,retrieved_at,evidence,verification_status,reviewed_by,reviewed_at"
      )
      .limit(10000),
  ]);
  if (companyError) return { ok: false, error: companyError.message };
  if (sourceError) return { ok: false, error: sourceError.message };
  const companyRows = (companies ?? []) as CompanyRow[];
  if (companyRows.length === 0) {
    return { ok: false, error: "There are no companies to score." };
  }
  const sourcesByCompany = new Map<string, SourceRow[]>();
  for (const source of (sources ?? []) as (SourceRow & {
    company_id: string;
  })[]) {
    const current = sourcesByCompany.get(source.company_id) ?? [];
    current.push(source);
    sourcesByCompany.set(source.company_id, current);
  }
  const batchTimestamp = new Date().toISOString();
  const scoreRows = companyRows.flatMap((company) => {
    const score = scoreCompany(sourcesByCompany.get(company.id) ?? []);
    return score.criteria.map((result) => ({
      company_id: company.id,
      criterion: result.criterion.key,
      weight: result.criterion.weight!,
      points: result.points ?? 0,
      reason: result.reason,
      source_id:
        result.points && result.sources.length > 0
          ? result.sources[0].id
          : null,
      created_at: batchTimestamp,
    }));
  });
  const { error: insertError } = await supabase
    .from("qualification_scores")
    .insert(scoreRows);
  if (insertError) return { ok: false, error: insertError.message };
  const audit = await writeAudit(supabase, {
    actorId: user.id,
    action: "qualification_scores_generated",
    entityType: "qualification_scores",
    detail: {
      companies_scored: companyRows.length,
      score_rows_created: scoreRows.length,
      scoring_batch: batchTimestamp,
      weights: Object.fromEntries(
        CRITERIA.map((criterion) => [criterion.key, criterion.weight])
      ),
      evidence_rule: "Only human-verified sources earn positive points.",
    },
  });
  revalidatePath("/prospects");
  for (const company of companyRows) {
    revalidatePath(/prospects/${company.id});
  }
  return {
    ok: true,
    companiesScored: companyRows.length,
    scoreRowsCreated: scoreRows.length,
    warning: audit.ok
      ? undefined
      : Scores were saved, but the audit entry failed: ${audit.error},
  };
}
''')
