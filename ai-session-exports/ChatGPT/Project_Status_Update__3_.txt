I've reviewed the main research flow. The structure is sound: URLs are validated before fetching, extra pages must belong to the homepage's hostname, pages are fetched sequentially, and AI claims are checked against fetched text before becoming CSV rows.
I found one small cache bug worth fixing before the first real run.
1. Fix cache freshness validation
Currently, a cache entry with a timestamp in the future can be accepted as fresh. Also, typeof fetchedAt === "number" accepts NaN and infinity.
Run this patch:

This accepts only finite timestamps from the past 24 hours.
2. One important security limitation
The URL checks reject IP-address literals and local hostnames, but Jina Reader performs the actual remote fetch. Our local validation cannot guarantee protection against DNS rebinding or unsafe redirects handled by that remote service. We should document this accurately rather than claim complete SSRF protection.

export function parseResearchInput(input: string): ResearchInputEntry[] {
  const entries: ResearchInputEntry[] = [];
  const errors: string[] = [];
  input.split(/\r?\n/).forEach((rawLine, index) => {
    const lineNumber = index + 1;
    const line = rawLine.trim();
if (!line || line.startsWith("#")) return;

const fields = line.split("|").map((field) => field.trim());
const [companyName, rawWebsite, ...rawExtraUrls] = fields;

if (!companyName || !rawWebsite) {
  errors.push(`Line ${lineNumber}: expected Company | Website | optional extra URLs`);
  return;
}

try {
  const website = validateHttpUrl(rawWebsite, lineNumber);
  const extraUrls = rawExtraUrls
    .filter(Boolean)
    .map((value) => validateHttpUrl(value, lineNumber));

  for (const extraUrl of extraUrls) {
    if (extraUrl.hostname !== website.hostname) {
      throw new Error(
        `Line ${lineNumber}: extra page must belong to the same website as the homepage`,
      );
    }
  }

  entries.push({
    companyName,
    website: website.toString(),
    extraUrls: [...new Set(extraUrls.map((url) => url.toString()))],
  });
} catch (error) {
  errors.push(
    error instanceof Error
      ? error.message
      : `Line ${lineNumber}: invalid entry`,
  );
}

  });
  if (errors.length) {
    throw new Error(Research input contains errors:\n${errors.join("\n")});
  }
  if (!entries.length) {
    throw new Error("Research input contains no companies");
  }
  return entries;
}
export async function runResearch(
  entries: ResearchInputEntry[],
  deps: RunnerDeps = {},
): Promise {
  const fetchPage = deps.fetchPage ?? fetchResearchPage;
  const extract = deps.extract ?? extractClaims;
  const discoverLinks = deps.discoverLinks ?? discoverSameSiteLinks;
  const now = deps.now ?? (() => new Date());
  const rows: ResearchOutputRow[] = [];
  const companies: CompanyResearchReport[] = [];
  for (const entry of entries) {
    const report: CompanyResearchReport = {
      companyName: entry.companyName,
      website: entry.website,
      pagesRead: [],
      proposed: 0,
      accepted: 0,
      rejected: 0,
      errors: [],
      rejectedClaims: [],
    };
const pages: ResearchPage[] = [];
const visited = new Set<string>();

const readPage = async (url: string) => {
  let safeUrl: URL;

  try {
    safeUrl = validateResearchUrl(url);

    const homepageHost = validateResearchUrl(entry.website).hostname;
    if (safeUrl.hostname !== homepageHost) {
      throw new Error("Page must belong to the same website as the homepage");
    }
  } catch (error) {
    report.errors.push(
      `Could not read ${url}: ${
        error instanceof Error ? error.message : String(error)
      }`,
    );
    return;
  }

  const normalizedUrl = safeUrl.toString();
  if (visited.has(normalizedUrl)) return;
  visited.add(normalizedUrl);

  try {
    const page = await fetchPage(normalizedUrl);
    pages.push(page);
    report.pagesRead.push(page.url);
  } catch (error) {
    report.errors.push(
      `Could not read ${url}: ${
        error instanceof Error ? error.message : String(error)
      }`,
    );
  }
};

// Read the homepage first so we can discover relevant same-site pages.
await readPage(entry.website);

const discovered = pages.length
  ? discoverLinks(pages[0].text, entry.website, 3)
  : [];

const additionalUrls = [...entry.extraUrls, ...discovered];

for (const url of additionalUrls) {
  await readPage(url);
}

if (!pages.length) {
  report.errors.push("No pages could be read; extraction was skipped");
  companies.push(report);
  continue;
}

let proposals: ProposedClaim[];

try {
  proposals = await extract(entry.companyName, pages);
  report.proposed = proposals.length;
} catch (error) {
  report.errors.push(
    `AI extraction failed: ${
      error instanceof Error ? error.message : String(error)
    }`,
  );
  companies.push(report);
  continue;
}

const verification = verifyClaims(proposals, pages);
const retrievedAt = now().toISOString().slice(0, 10);

report.accepted = verification.accepted.length;
report.rejected = verification.rejected.length;
report.rejectedClaims = verification.rejected.map((item) => ({
  claim_type: item.claim_type,
  reason: item.reason,
}));

for (const claim of verification.accepted) {
  rows.push({
    company_name: entry.companyName,
    website: entry.website,
    claim_type: claim.claim_type,
    claim: claim.claim,
    source_url: claim.source_url,
    source_title: claim.source_title,
    retrieved_at: retrievedAt,
    evidence: claim.evidence,
  });
}

companies.push(report);

  }
  return { rows, companies };
}

import { isIP } from "node:net";
import type { ResearchPage } from "./verify";
type FetchLike = (input: string | URL, init?: RequestInit) => Promise;
type PageDeps = {
  fetch?: FetchLike;
};
const MAX_PAGE_CHARACTERS = 120_000;
export function validateResearchUrl(input: string): URL {
  let url: URL;
  try {
    url = new URL(input);
  } catch {
    throw new Error("invalid URL");
  }
  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error("Only HTTP and HTTPS pages are supported");
  }
  if (url.username || url.password) {
    throw new Error("URLs containing credentials are not allowed");
  }
  const hostname = url.hostname.toLowerCase().replace(/.$/, "");
  const ipHostname =
    hostname.startsWith("[") && hostname.endsWith("]")
      ? hostname.slice(1, -1)
      : hostname;
  if (isIP(ipHostname) !== 0) {
    throw new Error("Private or local IP-address URLs are not allowed");
  }
  if (
    hostname === "localhost" ||
    hostname.endsWith(".localhost") ||
    hostname.endsWith(".local") ||
    hostname === "local"
  ) {
    throw new Error("Private or local URLs are not allowed");
  }
  url.hash = "";
  return url;
}
const safeHttpUrl = validateResearchUrl;
function fallbackTitle(url: URL): string {
  const path = decodeURIComponent(url.pathname)
    .split("/")
    .filter(Boolean)
    .pop();
  return path
    ? path.replace(/[-_]/g, " ").replace(/.[^.]+$/, "")
    : url.hostname;
}
export function parseReaderResponse(
  responseText: string,
  originalUrl: string,
): ResearchPage {
  const url = safeHttpUrl(originalUrl);
  const lines = responseText.replace(/\r\n/g, "\n").split("\n");
  const titleLine = lines.find((line) => /^Title:\s*/i.test(line));
  const title = titleLine?.replace(/^Title:\s*/i, "").trim();
  const markerIndex = lines.findIndex((line) =>
    /^Markdown Content:\s*$/i.test(line.trim()),
  );
  let contentLines =
    markerIndex >= 0 ? lines.slice(markerIndex + 1) : lines;
  // Remove Jina's metadata headers when the Markdown Content marker is absent.
  if (markerIndex < 0) {
    contentLines = contentLines.filter(
      (line) =>
        !/^(Title|URL Source|Markdown Content):\s*/i.test(line.trim()),
    );
  }
  const text = contentLines.join("\n").trim().slice(0, MAX_PAGE_CHARACTERS);
  if (!text) {
    throw new Error(Reader returned no page text for ${url.toString()});
  }
  return {
    url: url.toString(),
    title: title || fallbackTitle(url),
    text,
  };
}
const USEFUL_PATH = /about|company|our-story|team|leadership|contact|career|jobs|press|news|shipping|wholesale|retail|stockist/i;
export function discoverSameSiteLinks(
  markdown: string,
  baseUrl: string,
  maxLinks = 3,
): string[] {
  const base = safeHttpUrl(baseUrl);
  const found = new Map<string, { url: string; score: number }>();
  const markdownLink = /](([^)\s]+)(?:\s+"[^"]*")?)/g;
  let match: RegExpExecArray | null;
  while ((match = markdownLink.exec(markdown)) !== null) {
    const raw = match[1];
try {
  const candidate = new URL(raw, base);

  if (!["http:", "https:"].includes(candidate.protocol)) continue;
  if (candidate.hostname !== base.hostname) continue;
  if (candidate.username || candidate.password) continue;

  candidate.hash = "";
  candidate.search = "";

  // Skip the current page and common non-page assets.
  if (candidate.pathname === base.pathname) continue;
  if (/\.(pdf|jpg|jpeg|png|gif|webp|svg|css|js|zip|xml)$/i.test(candidate.pathname)) {
    continue;
  }

  const normalized = candidate.toString();
  const score = USEFUL_PATH.test(candidate.pathname) ? 1 : 0;

  if (!found.has(normalized)) {
    found.set(normalized, { url: normalized, score });
  }
} catch {
  // Ignore malformed links; one bad link should not stop page discovery.
}

  }
  return [...found.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.max(0, maxLinks))
    .map((item) => item.url);
}
export async function fetchResearchPage(
  originalUrl: string,
  deps: PageDeps = {},
): Promise {
  const target = safeHttpUrl(originalUrl);
  const fetcher = deps.fetch ?? fetch;
  const readerUrl = https://r.jina.ai/${target.toString()};
  const response = await fetcher(readerUrl, {
    method: "GET",
    headers: {
      Accept: "text/plain",
      "User-Agent": "LedgerwiseResearchPrototype/1.0",
    },
    signal: AbortSignal.timeout(30_000),
  });
  if (!response.ok) {
    throw new Error(
      Reader request failed for ${target.toString()}: HTTP ${response.status},
    );
  }
  const body = await response.text();
  return parseReaderResponse(body, target.toString());
}

const USEFUL_PATH = /about|company|our-story|team|leadership|contact|career|jobs|press|news|shipping|wholesale|retail|stockist/i;
export function discoverSameSiteLinks(
  markdown: string,
  baseUrl: string,
  maxLinks = 3,
): string[] {
  const base = safeHttpUrl(baseUrl);
  const found = new Map<string, { url: string; score: number }>();
  const markdownLink = /](([^)\s]+)(?:\s+"[^"]*")?)/g;
  let match: RegExpExecArray | null;
  while ((match = markdownLink.exec(markdown)) !== null) {
    const raw = match[1];
try {
  const candidate = new URL(raw, base);

  if (!["http:", "https:"].includes(candidate.protocol)) continue;
  if (candidate.hostname !== base.hostname) continue;
  if (candidate.username || candidate.password) continue;

  candidate.hash = "";
  candidate.search = "";

  // Skip the current page and common non-page assets.
  if (candidate.pathname === base.pathname) continue;
  if (/\.(pdf|jpg|jpeg|png|gif|webp|svg|css|js|zip|xml)$/i.test(candidate.pathname)) {
    continue;
  }

  const normalized = candidate.toString();
  const score = USEFUL_PATH.test(candidate.pathname) ? 1 : 0;

  if (!found.has(normalized)) {
    found.set(normalized, { url: normalized, score });
  }
} catch {
  // Ignore malformed links; one bad link should not stop page discovery.
}

  }
  return [...found.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.max(0, maxLinks))
    .map((item) => item.url);
}
export async function fetchResearchPage(
  originalUrl: string,
  deps: PageDeps = {},
): Promise {
  const target = safeHttpUrl(originalUrl);
  const fetcher = deps.fetch ?? fetch;
  const readerUrl = https://r.jina.ai/${target.toString()};
  const response = await fetcher(readerUrl, {
    method: "GET",
    headers: {
      Accept: "text/plain",
      "User-Agent": "LedgerwiseResearchPrototype/1.0",
    },
    signal: AbortSignal.timeout(30_000),
  });
  if (!response.ok) {
    throw new Error(
      Reader request failed for ${target.toString()}: HTTP ${response.status},
    );
  }
  const body = await response.text();
  return parseReaderResponse(body, target.toString());
}

7.2 — Preventing double bookings
This is one of the most important technical requirements.
We should not simply check whether a slot is free and then insert a booking. Two requests could arrive simultaneously.
Instead:

Use a database transaction or atomic database function.
Lock or constrain the requested slot.
Check whether it remains available.
Create the booking only if available.
Return a conflict if another booking won the race.

For Google synchronization, we also need:

An idempotency mechanism so retries don't create duplicate events.
A saved Google event ID.
A sync status for each booking.
Retry handling when Google is temporarily unavailable.

A booking must not be reported as fully confirmed until the required booking and calendar synchronization steps succeed. If synchronization fails, the application should show a pending or failed state and recover safely.
