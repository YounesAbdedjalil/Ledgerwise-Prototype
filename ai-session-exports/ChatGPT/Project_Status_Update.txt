import { CLAIM_TYPES, type ClaimType } from "../import-schema";

export type ResearchPage = {
  url: string;
  title: string;
  text: string;
};

export type ProposedClaim = {
  claim_type: string;
  claim: string;
  source_url: string;
  evidence: string;
};

export type VerifiedClaim = {
  claim_type: ClaimType;
  claim: string;
  source_url: string;
  source_title: string;
  evidence: string;
};

export type RejectedClaim = {
  index: number;
  claim_type: string;
  reason: string;
};

export type VerificationResult = {
  accepted: VerifiedClaim[];
  rejected: RejectedClaim[];
};

function normalizeText(value: string): string {
  return value
    .normalize("NFKC")
    .replace(/[‘’‛]/g, "'")
    .replace(/[“”„]/g, '"')
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ")
    .trim()
    .toLowerCase();
}

function normalizeUrl(value: string): string | null {
  try {
    const url = new URL(value);
    if (url.protocol !== "http:" && url.protocol !== "https:") return null;
    url.hash = "";
    return url.toString();
  } catch {
    return null;
  }
}

export function verifyClaims(
  proposals: ProposedClaim[],
  pages: ResearchPage[],
): VerificationResult {
  const pageByUrl = new Map<string, ResearchPage>();

  for (const page of pages) {
    const key = normalizeUrl(page.url);
    if (key) pageByUrl.set(key, page);
  }

  const accepted: VerifiedClaim[] = [];
  const rejected: RejectedClaim[] = [];
  const counts = new Map<ClaimType, number>();
  const seen = new Set<string>();

  proposals.forEach((proposal, index) => {
    const reject = (reason: string) => {
      rejected.push({
        index,
        claim_type: proposal.claim_type ?? "",
        reason,
      });
    };

    const claimType = proposal.claim_type as ClaimType;

    if (!CLAIM_TYPES.includes(claimType)) {
      reject("Unsupported claim type");
      return;
    }

    const claim = (proposal.claim ?? "").trim();
    const evidence = (proposal.evidence ?? "").trim();
    const sourceUrl = normalizeUrl(proposal.source_url ?? "");

    if (!claim || claim.length > 500) {
      reject("Claim is missing or exceeds 500 characters");
      return;
    }

    if (!sourceUrl) {
      reject("Source URL is not a valid HTTP(S) URL");
      return;
    }

    const page = pageByUrl.get(sourceUrl);
    if (!page) {
      reject("Source URL was not among the fetched pages");
      return;
    }

    if (!page.title.trim()) {
      reject("Fetched page has no title");
      return;
    }

    if (evidence.length < 20 || evidence.length > 300) {
      reject("Evidence must be between 20 and 300 characters");
      return;
    }

    const normalizedEvidence = normalizeText(evidence);
    const normalizedPage = normalizeText(page.text);

    if (!normalizedPage.includes(normalizedEvidence)) {
      reject("Evidence quote does not appear in the fetched page text");
      return;
    }

    if (claimType === "contact") {
      const parts = claim.split("|").map((part) => part.trim());
      const [name, role, email] = parts;

      if (!name || !role || parts.length > 3) {
        reject('Contact must use "Name | Role" or "Name | Role | email"');
        return;
      }

      if (
        !normalizedEvidence.includes(normalizeText(name)) ||
        !normalizedEvidence.includes(normalizeText(role))
      ) {
        reject("Contact name and role must both appear in the evidence quote");
        return;
      }

      if (email) {
        if (!/^\S+@\S+\.\S+$/.test(email)) {
          reject("Contact email is not a valid email address");
          return;
        }

        if (!normalizedPage.includes(normalizeText(email))) {
          reject("Contact email does not appear in the fetched page");
          return;
        }
      }
    }

    const duplicateKey = `${claimType}:${normalizeText(claim)}`;
    if (seen.has(duplicateKey)) {
      reject("Duplicate claim");
      return;
    }

    const count = counts.get(claimType) ?? 0;
    if (count >= 3) {
      reject("Maximum of three accepted claims per claim type reached");
      return;
    }

    seen.add(duplicateKey);
    counts.set(claimType, count + 1);

    accepted.push({
      claim_type: claimType,
      claim,
      source_url: page.url,
      source_title: page.title.trim(),
      evidence,
    });
  });

  return { accepted, rejected };
}

import type { ResearchPage } from "./verify";

type FetchLike = (input: string | URL, init?: RequestInit) => Promise<Response>;

type PageDeps = {
  fetch?: FetchLike;
};

const MAX_PAGE_CHARACTERS = 120_000;

function safeHttpUrl(input: string): URL {
  const url = new URL(input);

  if (!["http:", "https:"].includes(url.protocol)) {
    throw new Error("Only HTTP and HTTPS pages are supported");
  }

  if (url.username || url.password) {
    throw new Error("URLs containing credentials are not allowed");
  }

  if (
    url.hostname === "localhost" ||
    url.hostname.endsWith(".localhost") ||
    url.hostname.endsWith(".local") ||
    url.hostname === "127.0.0.1" ||
    url.hostname === "0.0.0.0" ||
    url.hostname.startsWith("10.") ||
    url.hostname.startsWith("192.168.") ||
    /^172\.(1[6-9]|2\d|3[01])\./.test(url.hostname)
  ) {
    throw new Error("Private or local URLs are not allowed");
  }

  url.hash = "";
  return url;
}

function fallbackTitle(url: URL): string {
  const path = decodeURIComponent(url.pathname)
    .split("/")
    .filter(Boolean)
    .pop();

  return path
    ? path.replace(/[-_]/g, " ").replace(/\.[^.]+$/, "")
    : url.hostname;
}

export function parseReaderResponse(
  responseText: string,
  originalUrl: string,
): ResearchPage {
  const url = safeHttpUrl(originalUrl);
  const lines = responseText.replace(/\r\n/g, "\n").split("\n");

  const titleLine = lines.find((line) => /^Title:\s*/i.test(line));
  const title = titleLine?.replace(/^Title:\s*/i, "").trim();

  const markerIndex = lines.findIndex((line) =>
    /^Markdown Content:\s*$/i.test(line.trim()),
  );

  let contentLines =
    markerIndex >= 0 ? lines.slice(markerIndex + 1) : lines;

  // Remove Jina's metadata headers when the Markdown Content marker is absent.
  if (markerIndex < 0) {
    contentLines = contentLines.filter(
      (line) =>
        !/^(Title|URL Source|Markdown Content):\s*/i.test(line.trim()),
    );
  }

  const text = contentLines.join("\n").trim().slice(0, MAX_PAGE_CHARACTERS);

  if (!text) {
    throw new Error(`Reader returned no page text for ${url.toString()}`);
  }

  return {
    url: url.toString(),
    title: title || fallbackTitle(url),
    text,
  };
}

const USEFUL_PATH = /about|company|our-story|team|leadership|contact|career|jobs|press|news|shipping|wholesale|retail|stockist/i;

export function discoverSameSiteLinks(
  markdown: string,
  baseUrl: string,
  maxLinks = 3,
): string[] {
  const base = safeHttpUrl(baseUrl);
  const found = new Map<string, { url: string; score: number }>();

  const markdownLink = /\]\(([^)\s]+)(?:\s+"[^"]*")?\)/g;
  let match: RegExpExecArray | null;

  while ((match = markdownLink.exec(markdown)) !== null) {
    const raw = match[1];

    try {
      const candidate = new URL(raw, base);

      if (!["http:", "https:"].includes(candidate.protocol)) continue;
      if (candidate.hostname !== base.hostname) continue;
      if (candidate.username || candidate.password) continue;

      candidate.hash = "";
      candidate.search = "";

      // Skip the current page and common non-page assets.
      if (candidate.pathname === base.pathname) continue;
      if (/\.(pdf|jpg|jpeg|png|gif|webp|svg|css|js|zip|xml)$/i.test(candidate.pathname)) {
        continue;
      }

      const normalized = candidate.toString();
      const score = USEFUL_PATH.test(candidate.pathname) ? 1 : 0;

      if (!found.has(normalized)) {
        found.set(normalized, { url: normalized, score });
      }
    } catch {
      // Ignore malformed links; one bad link should not stop page discovery.
    }
  }

  return [...found.values()]
    .sort((a, b) => b.score - a.score)
    .slice(0, Math.max(0, maxLinks))
    .map((item) => item.url);
}

export async function fetchResearchPage(
  originalUrl: string,
  deps: PageDeps = {},
): Promise<ResearchPage> {
  const target = safeHttpUrl(originalUrl);
  const fetcher = deps.fetch ?? fetch;
  const readerUrl = `https://r.jina.ai/${target.toString()}`;

  const response = await fetcher(readerUrl, {
    method: "GET",
    headers: {
      Accept: "text/plain",
      "User-Agent": "LedgerwiseResearchPrototype/1.0",
    },
    signal: AbortSignal.timeout(30_000),
  });

  if (!response.ok) {
    throw new Error(
      `Reader request failed for ${target.toString()}: HTTP ${response.status}`,
    );
  }

  const body = await response.text();
  return parseReaderResponse(body, target.toString());
}

import test from "node:test";
import assert from "node:assert/strict";
import {
  discoverSameSiteLinks,
  fetchResearchPage,
  parseReaderResponse,

test("parses a Jina Reader response and preserves the original source URL", () => {

      "URL Source: https://example.com/about",

  const page = parseReaderResponse(
    "Markdown Content:\nPage text with useful content.",
    "https://example.com/our-story",
  );

  assert.equal(page.title, "our story");
  assert.match(page.text, /useful content/);
});

test("rejects an empty Reader response", () => {
  assert.throws(
    () => parseReaderResponse("Title: Empty\nMarkdown Content:\n", "https://example.com"),
    /no page text/,
  );
});

test("discovers same-site links and prioritizes useful pages", () => {
  const markdown = [
    "[Home](https://example.com/)",
    "[Blog](https://example.com/blog)",
    "[About](https://example.com/about)",
    "[Careers](https://example.com/careers)",
    "[External](https://other.example/about)",
    "[PDF](https://example.com/files/report.pdf)",
  ].join("\n");

  const links = discoverSameSiteLinks(markdown, "https://example.com/", 3);

  assert.deepEqual(links, [
    "https://example.com/about",
    "https://example.com/careers",
    "https://example.com/blog",
  ]);
});

test("ignores duplicate links and removes tracking query strings", () => {
  const markdown = [
    "[About](https://example.com/about?utm_source=one)",
    "[About again](https://example.com/about?utm_source=two)",
  ].join("\n");

  const links = discoverSameSiteLinks(markdown, "https://example.com/");

  assert.deepEqual(links, ["https://example.com/about"]);
});

test("rejects non-HTTP and private/local URLs", () => {
  assert.throws(
    () => parseReaderResponse("Some text", "file:///etc/passwd"),
    /Only HTTP and HTTPS/,
  );

  assert.throws(

    /Private or local/,
  );

  assert.throws(

    /Private or local/,
  );

    fetch: async (input) => {
      requestedUrl = String(input);
      return new Response(
        [
          "Title: Example About",
          "URL Source: https://example.com/about",
          "Markdown Content:",
          "Example makes home goods for customers across the United States.",
        ].join("\n"),
        { status: 200 },
      );
    },
  });

  assert.equal(requestedUrl, "https://r.jina.ai/https://example.com/about");
  assert.equal(page.url, "https://example.com/about");
  assert.equal(page.title, "Example About");
});

test("reports Reader HTTP errors clearly", async () => {
  await assert.rejects(
    () =>
      fetchResearchPage("https://example.com", {
        fetch: async () => new Response("Rate limited", { status: 429 }),
      }),
    /HTTP 429/,
  );
});
