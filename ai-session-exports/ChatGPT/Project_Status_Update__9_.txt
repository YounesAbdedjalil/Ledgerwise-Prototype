-- 3. Serialize suppression with booking operations.
--    This MUST use the same advisory lock as the calendar RPCs:
--    pg_advisory_xact_lock(734921, 1).
create or replace function public.suppress_contact(
  p_email text,
  p_reason text,
  p_source_message_id uuid default null
)
returns public.suppressed_contacts
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_email text := lower(trim(coalesce(p_email, '')));
  v_row public.suppressed_contacts;
begin
  if auth.uid() is null then
    raise exception 'not_authenticated';
  end if;

  if v_email = ''
    or v_email !~ '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$'
  then
    raise exception 'invalid_email';
  end if;

  -- Shared transaction lock with book_slot and other calendar mutations.
  perform pg_advisory_xact_lock(734921, 1);

  insert into public.suppressed_contacts (
    email,
    reason,
    source_message_id
  )
  values (
    v_email,
    coalesce(nullif(trim(p_reason), ''), 'opt_out'),
    p_source_message_id
  )
  on conflict (lower(email)) do nothing
  returning * into v_row;

  if v_row.id is null then
    select *
      into v_row
      from public.suppressed_contacts
      where lower(email) = v_email;

    return v_row;
  end if;

  insert into public.audit_logs (
    actor_id,
    action,
    entity_type,
    entity_id,
    detail
  )
  values (
    auth.uid(),
    'contact_suppressed',
    'suppressed_contacts',
    v_row.id,
    jsonb_build_object(
      'email', v_email,
      'reason', v_row.reason,
      'source_message_id', p_source_message_id
    )
  );

  return v_row;
end;
$$;

revoke all on function public.suppress_contact(text, text, uuid)
  from public, anon, authenticated;

grant execute on function public.suppress_contact(text, text, uuid)
  to authenticated;

commit;
SQL

-- Ledgerwise migration 04

-- 1. Create calendar slot

create or replace function public.create_calendar_slot(
  p_start_time timestamptz,
  p_end_time timestamptz,
  p_timezone text
)
returns uuid
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_actor uuid := auth.uid();
  v_slot_id uuid;
  v_local_start timestamp;
  v_local_end timestamp;
begin
  if v_actor is null then
    raise exception 'Authentication required';
  end if;
  perform pg_advisory_xact_lock(734921, 1);
  if p_start_time is null or p_end_time is null
     or p_end_time <= p_start_time then
    raise exception 'Invalid slot time range';
  end if;
  if p_start_time <= now() then
    raise exception 'A calendar slot must be in the future';
  end if;
  if p_timezone is null or not exists (
    select 1 from pg_timezone_names where name = p_timezone
  ) then
    raise exception 'Invalid timezone';
  end if;
  if extract(epoch from (p_end_time - p_start_time)) / 60 not between 15 and 120 then
    raise exception 'Slot duration must be between 15 and 120 minutes';
  end if;
  v_local_start := p_start_time at time zone p_timezone;
  v_local_end := p_end_time at time zone p_timezone;
  if v_local_start::date <> v_local_end::date
     or v_local_start::time < time '09:00'
     or v_local_end::time > time '17:00' then
    raise exception 'Slots must remain within 09:00-17:00 local business hours';
  end if;
  if exists (
    select 1
    from public.calendar_slots s
    where s.is_available = true
      and s.start_time < p_end_time
      and s.end_time > p_start_time
  ) then
    raise exception 'This slot overlaps an existing available slot';
  end if;
  insert into public.calendar_slots (
    start_time,
    end_time,
    timezone,
    source,
    external_id,
    is_available,
    fetched_at
  )
  values (
    p_start_time,
    p_end_time,
    p_timezone,
    'internal',
    null,
    true,
    now()
  )
  returning id into v_slot_id;
  insert into public.audit_logs (
    actor_id, action, entity_type, entity_id, detail
  )
  values (
    v_actor,
    'calendar_slot_created',
    'calendar_slot',
    v_slot_id,
    jsonb_build_object(
      'start_time', p_start_time,
      'end_time', p_end_time,
      'timezone', p_timezone
    )
  );
  return v_slot_id;
end;
$$;

-- 2. Delete calendar slot

create or replace function public.delete_calendar_slot(
  p_slot_id uuid
)
returns boolean
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_actor uuid := auth.uid();
  v_slot public.calendar_slots%rowtype;
begin
  if v_actor is null then
    raise exception 'Authentication required';
  end if;
  if p_slot_id is null then
    raise exception 'Slot ID is required';
  end if;
  perform pg_advisory_xact_lock(734921, 1);
  select *
  into v_slot
  from public.calendar_slots
  where id = p_slot_id
  for update;
  if not found then
    raise exception 'Calendar slot not found';
  end if;
  if exists (
    select 1
    from public.bookings b
    where b.slot_id = p_slot_id
      and b.status in ('pending', 'confirmed', 'needs_human')
  ) then
    raise exception 'Cannot delete a slot with an active booking';
  end if;
  if exists (
    select 1
    from public.bookings b
    where b.slot_id = p_slot_id
  ) then
    raise exception 'Cannot delete a slot referenced by booking history';
  end if;
  delete from public.calendar_slots
  where id = p_slot_id;
  insert into public.audit_logs (
    actor_id, action, entity_type, entity_id, detail
  )
  values (
    v_actor,
    'calendar_slot_deleted',
    'calendar_slot',
    p_slot_id,
    jsonb_build_object(
      'start_time', v_slot.start_time,
      'end_time', v_slot.end_time,
      'timezone', v_slot.timezone
    )
  );
  return true;
end;
$$;

-- 6. Reschedule booking

create or replace function public.reschedule_booking(
  p_booking_id uuid,
  p_new_slot_id uuid,
  p_reason text
)
returns boolean
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_actor uuid := auth.uid();
  v_booking public.bookings%rowtype;
  v_new_slot public.calendar_slots%rowtype;
  v_reason text := nullif(btrim(coalesce(p_reason, '')), '');
  v_old_slot_id uuid;
begin
  if v_actor is null then
    raise exception 'Authentication required';
  end if;
  if p_booking_id is null or p_new_slot_id is null then
    raise exception 'Booking ID and new slot ID are required';
  end if;
  if v_reason is null then
    raise exception 'A rescheduling reason is required';
  end if;
  perform pg_advisory_xact_lock(734921, 1);
  select *
  into v_booking
  from public.bookings
  where id = p_booking_id
  for update;
  if not found then
    raise exception 'Booking not found';
  end if;
  if v_booking.status not in ('pending', 'confirmed', 'needs_human') then
    raise exception 'This booking cannot be rescheduled from its current status';
  end if;
  if v_booking.slot_id = p_new_slot_id then
    raise exception 'Choose a different slot';
  end if;
  if exists (
    select 1
    from public.suppressed_contacts s
    where lower(s.email) = lower(v_booking.prospect_email)
  ) then
    raise exception 'This contact is suppressed';
  end if;
  v_old_slot_id := v_booking.slot_id;
  -- Lock both slot rows in stable ID order.
  perform s.id
  from public.calendar_slots s
  where s.id in (v_old_slot_id, p_new_slot_id)
  order by s.id
  for update;
  select *
  into v_new_slot
  from public.calendar_slots
  where id = p_new_slot_id;
  if not found then
    raise exception 'New calendar slot not found';
  end if;
  if not v_new_slot.is_available or v_new_slot.start_time <= now() then
    raise exception 'New slot is not available';
  end if;
  if exists (
    select 1
    from public.bookings b
    join public.calendar_slots s on s.id = b.slot_id
    where b.id <> p_booking_id
      and b.status in ('pending', 'confirmed', 'needs_human')
      and s.start_time < v_new_slot.end_time
      and s.end_time > v_new_slot.start_time
  ) then
    raise exception 'New slot overlaps an existing booking';
  end if;
  update public.bookings
  set slot_id = p_new_slot_id,
      status = 'pending'
  where id = p_booking_id;
  insert into public.audit_logs (
    actor_id, action, entity_type, entity_id, detail
  )
  values (
    v_actor,
    'booking_rescheduled',
    'booking',
    p_booking_id,
    jsonb_build_object(
      'old_slot_id', v_old_slot_id,
      'new_slot_id', p_new_slot_id,
      'previous_status', v_booking.status,
      'reason', v_reason
    )
  );
  return true;
end;
$$;

async function checkSlotFree(
  supabase: ServerSupabase,
  slot: SlotRow,
  ignoreBookingId?: string
): Promise<{ ok: boolean; error?: string }> {
  const { data: overlappingSlots, error: slotError } = await supabase
    .from("calendar_slots")
    .select("id")
    .lt("start_time", slot.end_time)
    .gt("end_time", slot.start_time);
  if (slotError) return { ok: false, error: slotError.message };
  const slotIds = Array.from(
    new Set([slot.id, ...(overlappingSlots ?? []).map((row) => row.id)])
  );
  const { data, error } = await supabase
    .from("bookings")
    .select("id")
    .in("slot_id", slotIds)
    .in("status", [...OCCUPYING_BOOKING_STATUSES]);
  if (error) return { ok: false, error: error.message };
  const conflict = (data ?? []).find((row) => row.id !== ignoreBookingId);

    return { ok: false, error: "That time is already held by another booking." };

export async function deleteSlot(slotId: string): Promise {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);
  if (!user) return { ok: false, error: "Not signed in." };
  if (!isUuid(slotId)) return { ok: false, error: "Invalid slot ID." };
  const loaded = await loadSlot(supabase, slotId);
  if (!loaded.slot) return { ok: false, error: loaded.error };
  const { data: active, error } = await supabase
    .from("bookings")
    .select("id")
    .eq("slot_id", slotId)
    .in("status", [...OCCUPYING_BOOKING_STATUSES])
    .limit(1);
  if (error) return { ok: false, error: error.message };
  if ((active ?? []).length > 0) {
    return {
      ok: false,
      error: "Cancel or reschedule the active booking before deleting this slot.",
    };
  }
  const { error: deleteError } = await supabase
    .from("calendar_slots")
    .delete()
    .eq("id", slotId);
  if (deleteError) return { ok: false, error: deleteError.message };
  await writeAudit(supabase, {
    actorId: user.id,
    action: "calendar_slot_deleted",
    entityType: "calendar_slots",
    entityId: slotId,
    detail: {
      start_time: loaded.slot.start_time,
      end_time: loaded.slot.end_time,
    },
  });
  revalidateCalendar();
  return { ok: true, slotId, message: "Slot deleted." };
}

  bookingId: string,

  reason: string

  const user = await getSessionUser(supabase);

  if (!loaded.booking) return { ok: false, error: loaded.error };
  const { booking } = loaded;
  if (
    booking.status !== "pending" &&
    booking.status !== "confirmed" &&
    booking.status !== "needs_human"
  ) {
    return { ok: false, error: "Only an active booking can be rescheduled." };
  }
  if (booking.slot_id === newSlotId) {
    return { ok: false, error: "Choose a different slot." };
  }

      ok: false,

  if (!free.ok) return { ok: false, error: free.error };

    .select("id")
    .maybeSingle();

      return { ok: false, error: "That slot was just taken. Refresh and choose another." };
    }

  }

    return {
      ok: false,
      error: "The booking changed while you were rescheduling it. Refresh and try again.",
    };
  }

        booking.google_event_id,
        bookingEventInput(booking, loadedSlot.slot, loadedSlot.slot.timezone)

    : await syncBookingToGoogle(
        bookingEventInput(booking, loadedSlot.slot, loadedSlot.slot.timezone)
      );

  }
  const syncNote =
    google.status === "synced"
      ? " Google Calendar was updated."
      : google.status === "failed"
        ? " Google Calendar sync failed; the internal booking is authoritative."
        : booking.google_event_id
          ? " The Google event was not updated because sync is not operational."
          : "";
  await writeAudit(supabase, {
    actorId: user.id,
    action: "booking_rescheduled",
    entityType: "bookings",

      new_slot_id: newSlotId,
      previous_status: booking.status,
      reason: cleanReason,
      google_sync: google.status,
    },
  });
  revalidateCalendar();
  return {
    ok: true,
    bookingId,
    slotId: newSlotId,
    message: Booking rescheduled. The old slot is free and the new slot is held.${syncNote},
  };
}

export async function confirmBooking(
  bookingId: string
): Promise<CalendarActionResult> {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);

  if (!user) return { ok: false, error: "Not signed in." };
  if (!isUuid(bookingId)) return { ok: false, error: "Invalid booking ID." };

  const loaded = await loadBooking(supabase, bookingId);
  if (!loaded.booking) {
    return { ok: false, error: loaded.error };
  }

  const booking = loaded.booking;

  if (booking.status !== "pending" && booking.status !== "needs_human") {
    return {
      ok: false,
      error: "Only a pending or needs-human booking can be confirmed.",
    };
  }

  const suppression = await checkSuppressed(supabase, booking.prospect_email);
  if (!suppression.ok) {
    return {
      ok: false,
      error: `${suppression.error} Cancel the booking instead.`,
    };
  }

  const { data, error } = await supabase.rpc("confirm_booking", {
    p_booking_id: bookingId,
  });

  if (error || data !== true) {
    return {
      ok: false,
      error: rpcError(error, "The booking could not be confirmed."),
    };
  }

  let syncNote = "";

  if (!booking.google_event_id && booking.slot_id) {
    const loadedSlot = await loadSlot(supabase, booking.slot_id);

    if (loadedSlot.slot) {
      const google = await syncBookingToGoogle(
        bookingEventInput(booking, loadedSlot.slot, loadedSlot.slot.timezone)
      );

      const metadataSaved = await recordGoogleSync(bookingId, google);

      syncNote =
        google.status === "synced"
          ? metadataSaved
            ? " Google Calendar event created."
            : " Google event created, but its sync metadata could not be saved."
          : google.status === "failed"
            ? " Google Calendar sync failed; the internal booking is still confirmed."
            : "";
    }
  }

  revalidateCalendar();

  return {
    ok: true,
    bookingId,
    message: `Booking confirmed.${syncNote}`,
  };
}

export async function cancelBooking(
  bookingId: string,
  reason: string
): Promise<CalendarActionResult> {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);

  if (!user) return { ok: false, error: "Not signed in." };
  if (!isUuid(bookingId)) return { ok: false, error: "Invalid booking ID." };

  const cleanReason = trimText(reason, 500);
  if (!cleanReason) {
    return { ok: false, error: "A cancellation reason is required." };
  }

  const loaded = await loadBooking(supabase, bookingId);
  if (!loaded.booking) {
    return { ok: false, error: loaded.error };
  }

  const booking = loaded.booking;

  if (
    booking.status !== "pending" &&
    booking.status !== "confirmed" &&
    booking.status !== "needs_human"
  ) {
    return {
      ok: false,
      error: "This booking is already cancelled or rescheduled.",
    };
  }

  const { data, error } = await supabase.rpc("cancel_booking", {
    p_booking_id: bookingId,
    p_reason: cleanReason,
  });

  if (error || data !== true) {
    return {
      ok: false,
      error: rpcError(error, "The booking could not be cancelled."),
    };
  }

  let syncNote = "";

  if (booking.google_event_id) {
    const google = await deleteGoogleEvent(booking.google_event_id);

    if (google.status === "not_configured") {
      const metadataSaved = await recordGoogleSync(
        bookingId,
        google,
        false,
        true
      );

      syncNote = metadataSaved
        ? " The internal booking is cancelled, but the Google event could not be removed because sync is not operational."
        : " The internal booking is cancelled, but Google sync metadata could not be updated.";
    } else {
      const metadataSaved = await recordGoogleSync(bookingId, google, true);

      syncNote =
        google.status === "synced"
          ? metadataSaved
            ? " The Google Calendar event was removed."
            : " The Google event was removed, but sync metadata could not be saved."
          : " The internal booking is cancelled, but Google Calendar deletion failed.";
    }
  }

  revalidateCalendar();

  return {
    ok: true,
    bookingId,
    message: `Booking cancelled. The slot is free.${syncNote}`,
  };
}

export async function rescheduleBooking(
  bookingId: string,
  newSlotId: string,
  reason: string
): Promise<CalendarActionResult> {
  const supabase = await createClient();
  const user = await getSessionUser(supabase);

  if (!user) return { ok: false, error: "Not signed in." };

  if (!isUuid(bookingId) || !isUuid(newSlotId)) {
    return { ok: false, error: "Invalid booking or slot ID." };
  }

  const cleanReason = trimText(reason, 500);
  if (!cleanReason) {
    return { ok: false, error: "A reschedule reason is required." };
  }

  const loaded = await loadBooking(supabase, bookingId);
  if (!loaded.booking) {
    return { ok: false, error: loaded.error };
  }

  const booking = loaded.booking;

  if (
    booking.status !== "pending" &&
    booking.status !== "confirmed" &&
    booking.status !== "needs_human"
  ) {
    return {
      ok: false,
      error: "Only an active booking can be rescheduled.",
    };
  }

  if (booking.slot_id === newSlotId) {
    return { ok: false, error: "Choose a different slot." };
  }

  const suppression = await checkSuppressed(supabase, booking.prospect_email);
  if (!suppression.ok) {
    return {
      ok: false,
      error: `${suppression.error} Cancel the booking instead.`,
    };
  }

  const loadedSlot = await loadSlot(supabase, newSlotId);
  if (!loadedSlot.slot) {
    return { ok: false, error: loadedSlot.error };
  }

  const { data, error } = await supabase.rpc("reschedule_booking", {
    p_booking_id: bookingId,
    p_new_slot_id: newSlotId,
    p_reason: cleanReason,
  });

  if (error || data !== true) {
    return {
      ok: false,
      error: rpcError(error, "The booking could not be rescheduled."),
    };
  }

  const google = booking.google_event_id
    ? await updateGoogleEvent(
        booking.google_event_id,
        bookingEventInput(booking, loadedSlot.slot, loadedSlot.slot.timezone)
      )
    : await syncBookingToGoogle(
        bookingEventInput(booking, loadedSlot.slot, loadedSlot.slot.timezone)
      );

  let metadataSaved = true;

  if (google.status === "not_configured" && booking.google_event_id) {
    metadataSaved = await recordGoogleSync(bookingId, google, false, true);
  } else {
    metadataSaved = await recordGoogleSync(bookingId, google);
  }

  const syncNote =
    google.status === "synced"
      ? metadataSaved
        ? " Google Calendar was updated."
        : " Google Calendar was updated, but its sync metadata could not be saved."
      : google.status === "failed"
        ? " Google Calendar sync failed; the internal booking is authoritative."
        : booking.google_event_id
          ? " The Google event was not updated because sync is not operational."
          : "";

  revalidateCalendar();

  return {
    ok: true,
    bookingId,
    slotId: newSlotId,
    message: `Booking rescheduled. The old slot is free and the new slot is held.${syncNote}`,
  };
}

3. Three important recommendations
A. Distinguish facts from interpretations.
For example, Sigmas' transition from a marketplace to DTC is a factual business development.
But saying that this transition creates customer acquisition pressure is an interpretation. That's a reasonable hypothesis, but don't present it as a problem the company has publicly acknowledged.
B. Make sources traceable.
Ideally, every claim should have its own supporting URL. If a claim contains five different statistics, make sure the source actually supports all five.
C. Clean up the claim formatting.
Your Wild Rye growth_signal and pain_signal contain Markdown formatting such as **bold**. I'd remove that formatting and keep the CSV values as plain text. Also, avoid putting several unrelated contact details into a single hard-to-verify statement.
4. About the two empty columns
Your decision to leave retrieved_at empty is understandable if you intend to populate it later.
For evidence, though, I would reconsider leaving it empty permanently. The distinction I'd recommend is:

claim: A concise statement of what the research establishes.
source_url: Where the information came from.
source_title: The name of the source.
retrieved_at: When you collected the information.
evidence: The specific supporting detail, quotation, or excerpt from the source.

This makes the data much easier to audit and helps prevent the AI from turning weak signals into confident claims.
