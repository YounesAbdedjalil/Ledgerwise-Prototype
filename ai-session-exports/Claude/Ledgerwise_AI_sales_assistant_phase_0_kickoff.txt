import { createServerClient } from "@supabase/ssr";
import { cookies } from "next/headers";

export async function createClient() {
  const cookieStore = await cookies();
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return cookieStore.getAll();
        },
        setAll(list) {
          try {
            list.forEach(({ name, value, options }) =>
              cookieStore.set(name, value, options)
            );
          } catch {
            // Called from a Server Component; the middleware refreshes sessions.
          }
        },
      },
    }
  );
}

cat > middleware.ts <<'EOF'
import { createServerClient } from "@supabase/ssr";
import { NextResponse, type NextRequest } from "next/server";

export async function middleware(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll() {
          return request.cookies.getAll();
        },
        setAll(list) {
          list.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          list.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          );
        },
      },
    }
  );

  const {
    data: { user },
  } = await supabase.auth.getUser();
  const path = request.nextUrl.pathname;

  if (!user && path !== "/login") {
    const url = request.nextUrl.clone();
    url.pathname = "/login";
    return NextResponse.redirect(url);
  }
  if (user && path === "/login") {
    const url = request.nextUrl.clone();
    url.pathname = "/prospects";
    return NextResponse.redirect(url);
  }
  return response;
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico).*)"],
};
EOF

"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { createClient } from "@/lib/supabase/client";

export default function LoginPage() {
  const router = useRouter();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setLoading(true);
    setError(null);
    const supabase = createClient();
    const { error } = await supabase.auth.signInWithPassword({ email, password });
    if (error) {
      setError(error.message);
      setLoading(false);
      return;
    }
    router.push("/prospects");
    router.refresh();
  }

  return (
    <main className="min-h-screen flex items-center justify-center bg-gray-50">
      <form
        onSubmit={handleSubmit}
        className="w-full max-w-sm bg-white p-8 rounded-lg shadow space-y-4"
      >
        <h1 className="text-2xl font-semibold">Ledgerwise Prototype</h1>
        <p className="text-sm text-gray-500">Sign in to the sales assistant.</p>
        <input
          type="email"
          required
          placeholder="Email"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
          className="w-full border rounded px-3 py-2"
        />
        <input
          type="password"
          required
          placeholder="Password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          className="w-full border rounded px-3 py-2"
        />
        {error && <p className="text-sm text-red-600">{error}</p>}
        <button
          type="submit"
          disabled={loading}
          className="w-full bg-blue-600 text-white rounded py-2 disabled:opacity-50"
        >
          {loading ? "Signing in..." : "Sign in"}
        </button>
      </form>
    </main>
  );
}

Claude:
Cloudflare vs Vercel for this project
I'd stick with Vercel. Cloudflare is a good platform, but it doesn't fit a 72-hour build of a Next.js app.
Why Vercel fits better

Next.js is Vercel's own framework. Middleware, server components, route handlers and environment variables work with no adaptation. Our Phase 0 code (Supabase auth middleware, server actions, the signout route) will deploy as written.
Cloudflare runs Next.js through an adapter (OpenNext for Cloudflare, running on Workers). It works, but it adds config files, a different build and deploy command, and a runtime that isn't standard Node. Supabase's SSR auth, and the AI SDK calls and Cal.com requests we'll add later, are where compatibility surprises usually appear. I haven't verified the adapter's current state, and Cloudflare's Next.js support has changed a lot recently, so I'd have to check its docs before every step.
Debugging cost. With a hard deadline, an hour lost to a Workers-specific error is an hour not spent on research, guardrails or the test set.
Your spec names Vercel. You're allowed to deviate for a materially better alternative, and Cloudflare isn't one here. A deviation would also need justifying in HOW_I_WORKED.md.

Where Cloudflare would be better

Cheaper or more generous limits at scale
Edge performance, DDoS protection and DNS in one place
You already use Cloudflare and prefer it

None of that matters for a prototype with a handful of sales users.
My recommendation
Use Vercel now. Vercel's free Hobby plan is meant for personal, non-commercial use, which covers a prototype. If Ledgerwise ever ran this commercially, you'd move to a paid plan or migrate, and I can help with that then.
If you have a specific reason for Cloudflare (an existing account, a cost concern, or a requirement I don't know about), tell me and I'll adapt the deployment steps. I'd first check Cloudflare's current Next.js docs for exact commands rather than guess, and our Supabase code would stay the same.
Otherwise, carry on with Steps 1 to 6. Let me know which gate items pass or fail, and paste any error text exactly.

-- ============ USERS (profile mirror of auth.users) ============
create table public.users (
  id uuid primary key references auth.users(id) on delete cascade,
  email text not null,
  full_name text,
  role text not null default 'sales',
  created_at timestamptz not null default now()
);

create or replace function public.handle_new_user()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  insert into public.users (id, email) values (new.id, new.email)
  on conflict (id) do nothing;
  return new;
end $$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- backfill the user you already created
insert into public.users (id, email)
select id, email from auth.users on conflict (id) do nothing;

-- ============ COMPANIES ============
create table public.companies (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  website text not null,
  category text not null default 'unknown',
  us_ecommerce text not null default 'unknown'
    check (us_ecommerce in ('verified','unknown')),
  contact_name text not null default 'unknown',
  contact_role text not null default 'unknown',
  contact_email text,            -- only if publicly listed; else null
  growth_signal text not null default 'unknown',
  research_notes text,
  total_score integer not null default 0,
  qualification_status text not null default 'new'
    check (qualification_status in ('new','qualified','not_qualified','needs_review')),
  spot_checked boolean not null default false,
  created_at timestamptz not null default now()
);

-- ============ COMPANY SOURCES (every material fact) ============
create table public.company_sources (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  claim text not null,                       -- the fact being supported
  claim_type text not null
    check (claim_type in ('us_ecommerce','category','contact','growth_signal','size','other')),
  source_url text not null check (source_url ~* '^https?://'),
  source_title text not null,
  retrieved_at date not null,
  evidence text not null,                    -- short quote or concise summary
  verified_by uuid references public.users(id),
  created_at timestamptz not null default now()
);

-- ============ QUALIFICATION SCORES ============
create table public.qualification_scores (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  criterion text not null,
  weight integer not null check (weight > 0),
  points integer not null check (points >= 0),
  reason text not null,
  source_id uuid references public.company_sources(id) on delete set null,
  created_at timestamptz not null default now(),
  check (points <= weight),
  -- any positive score must be traceable to a source
  check (points = 0 or source_id is not null)
);

-- ============ EMAIL DRAFTS / EVENTS ============
create table public.email_drafts (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  subject text not null,
  body text not null,
  status text not null default 'draft'
    check (status in ('draft','needs_review','approved','rejected','sent_simulated','sent')),
  source_ids uuid[] not null default '{}',
  revision_note text,
  reviewed_by uuid references public.users(id),
  reviewed_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create table public.email_events (
  id uuid primary key default gen_random_uuid(),
  draft_id uuid not null references public.email_drafts(id) on delete cascade,
  event_type text not null,
  actor_id uuid references public.users(id),
  detail jsonb not null default '{}',
  created_at timestamptz not null default now()
);

-- ============ CONVERSATIONS / MESSAGES ============
create table public.conversations (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies(id) on delete cascade,
  contact_email text not null,
  status text not null default 'ai_handling'
    check (status in ('ai_handling','needs_human','human_takeover','closed','suppressed')),
  assigned_to uuid references public.users(id),
  created_at timestamptz not null default now()

create table public.messages (
  id uuid primary key default gen_random_uuid(),

  body text not null,
  classification text,

  guardrail_flags text[] not null default '{}',
  is_simulated boolean not null default true,
  created_at timestamptz not null default now()

  source text not null default 'calcom' check (source in ('calcom','demo')),

-- ============ IMMUTABILITY: audit log + suppression list ============
create or replace function public.block_changes()
returns trigger language plpgsql as $$
begin
  raise exception '% on % is not allowed (append-only table)', tg_op, tg_table_name;
end $$;

create trigger audit_logs_append_only
  before update or delete on public.audit_logs
  for each row execute function public.block_changes();

create trigger suppressed_permanent
  before update or delete on public.suppressed_contacts
  for each row execute function public.block_changes();

-- ============ ROW-LEVEL SECURITY ============
do $$
declare t text;
begin
  foreach t in array array[
    'users','companies','company_sources','qualification_scores',
    'email_drafts','email_events','conversations','messages',
    'calendar_slots','bookings','audit_logs','suppressed_contacts'
  ] loop
    execute format('alter table public.%I enable row level security', t);
    execute format('create policy "auth read" on public.%I for select to authenticated using (true)', t);
  end loop;

  foreach t in array array[
    'companies','company_sources','qualification_scores','email_drafts',
    'email_events','conversations','messages','calendar_slots','bookings',
    'audit_logs','suppressed_contacts'
  ] loop
    execute format('create policy "auth insert" on public.%I for insert to authenticated with check (true)', t);
  end loop;

  foreach t in array array[
    'companies','company_sources','qualification_scores','email_drafts',
    'conversations','messages','calendar_slots','bookings'
  ] loop
    execute format('create policy "auth update" on public.%I for update to authenticated using (true) with check (true)', t);
    execute format('create policy "auth delete" on public.%I for delete to authenticated using (true)', t);
  end loop;
end $$;

export const CLAIM_TYPES = [
  "us_ecommerce",
  "category",
  "contact",
  "growth_signal",
  "size",
  "other",
] as const;
export type ClaimType = (typeof CLAIM_TYPES)[number];

export const REQUIRED_COLUMNS = [
  "company_name",
  "website",
  "claim_type",
  "claim",
  "source_url",
  "source_title",
  "retrieved_at",
  "evidence",
] as const;

export type RawRow = Record<string, string>;

export type ValidRow = {
  company_name: string;
  host: string; // normalized hostname, used to deduplicate companies
  claim_type: ClaimType;
  claim: string;
  source_url: string;
  source_title: string;
  retrieved_at: string; // YYYY-MM-DD
  evidence: string;
};

export type RowResult =
  | { ok: true; row: ValidRow }
  | { ok: false; errors: string[] };

export function hostFromUrl(input: string): string | null {
  try {
    const u = new URL(/^https?:\/\//i.test(input) ? input : `https://${input}`);
    if (!u.hostname.includes(".")) return null;
    return u.hostname.replace(/^www\./, "").toLowerCase();
  } catch {
    return null;
  }
}

export function validateRow(raw: RawRow): RowResult {
  const errors: string[] = [];
  const get = (k: string) => (raw[k] ?? "").trim();

  const company_name = get("company_name");
  if (!company_name) errors.push("company_name is missing");

  const host = hostFromUrl(get("website"));
  if (!host) errors.push("website is missing or not a valid URL");

  const claim_type = get("claim_type") as ClaimType;
  if (!CLAIM_TYPES.includes(claim_type))
    errors.push(`claim_type must be one of: ${CLAIM_TYPES.join(", ")}`);

  const claim = get("claim");
  if (!claim) errors.push("claim is missing");

  const source_url = get("source_url");
  if (!/^https?:\/\/\S+\.\S+/i.test(source_url))
    errors.push("source_url must be a full http(s) URL");

  const source_title = get("source_title");
  if (!source_title) errors.push("source_title is missing");

  const retrieved_at = get("retrieved_at");
  const today = new Date().toISOString().slice(0, 10);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(retrieved_at) || isNaN(Date.parse(retrieved_at))) {
    errors.push("retrieved_at must be a real date as YYYY-MM-DD");
  } else if (retrieved_at > today) {
    errors.push("retrieved_at cannot be in the future");
  }

  const evidence = get("evidence");
  if (evidence.length < 10)
    errors.push("evidence must be a quote or summary of at least 10 characters");

  if (claim_type === "contact") {
    const parts = claim.split("|").map((p) => p.trim());
    if (parts.length < 2 || !parts[0] || !parts[1])
      errors.push('contact claim must look like "Name | Role" or "Name | Role | email"');
    if (parts[2] && !/^\S+@\S+\.\S+$/.test(parts[2]))
      errors.push("contact email in claim is not a valid email");
  }

  if (errors.length > 0 || !host) return { ok: false, errors };
  return {
    ok: true,
    row: { company_name, host, claim_type, claim, source_url, source_title, retrieved_at, evidence },
  };
}

"use server";

import { createClient } from "@/lib/supabase/server";
import { validateRow, type RawRow, type ValidRow } from "@/lib/import-schema";

export type ImportResult = {
  error?: string;
  rejected: { line: number; errors: string[] }[];
  companiesCreated: number;
  sourcesAdded: number;
  duplicatesSkipped: number;
};

const empty: ImportResult = {
  rejected: [],
  companiesCreated: 0,
  sourcesAdded: 0,
  duplicatesSkipped: 0,
};

export async function importRows(rows: RawRow[]): Promise<ImportResult> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) return { ...empty, error: "Not signed in." };
  if (rows.length === 0) return { ...empty, error: "The file has no rows." };
  if (rows.length > 2000) return { ...empty, error: "Max 2000 rows per import." };

  // Authoritative validation happens here, on the server.
  const result: ImportResult = { ...empty, rejected: [] };
  const valid: ValidRow[] = [];
  rows.forEach((raw, i) => {
    const r = validateRow(raw);
    if (r.ok) valid.push(r.row);
    else result.rejected.push({ line: i + 2, errors: r.errors }); // +2: header is line 1
  });

  const groups = new Map<string, ValidRow[]>();
  for (const row of valid) {
    groups.set(row.host, [...(groups.get(row.host) ?? []), row]);
  }

  for (const [host, groupRows] of groups) {
    const website = `https://${host}`;

    let { data: company } = await supabase
      .from("companies")
      .select("id, category, growth_signal, contact_name, contact_role, contact_email")
      .eq("website", website)
      .maybeSingle();

    if (!company) {
      const { data: created, error } = await supabase
        .from("companies")
        .insert({ name: groupRows[0].company_name, website })
        .select("id, category, growth_signal, contact_name, contact_role, contact_email")
        .single();
      if (error || !created) {
        result.rejected.push({
          line: 0,
          errors: [`Could not create ${host}: ${error?.message ?? "unknown error"}`],
        });
        continue;
      }
      company = created;
      result.companiesCreated++;
    }

    const { data: existing } = await supabase
      .from("company_sources")
      .select("claim, source_url")
      .eq("company_id", company.id);
    const seen = new Set((existing ?? []).map((s) => `${s.source_url}||${s.claim}`));

    const toInsert = groupRows
      .filter((r) => {
        const key = `${r.source_url}||${r.claim}`;
        if (seen.has(key)) {
          result.duplicatesSkipped++;
          return false;
        }
        seen.add(key);
        return true;
      })
      .map((r) => ({
        company_id: company!.id,
        claim: r.claim,
        claim_type: r.claim_type,
        source_url: r.source_url,
        source_title: r.source_title,
        retrieved_at: r.retrieved_at,
        evidence: r.evidence,
        verified_by: user.id,
      }));

    if (toInsert.length > 0) {
      const { error } = await supabase.from("company_sources").insert(toInsert);
      if (error) {
        result.rejected.push({ line: 0, errors: [`Sources for ${host}: ${error.message}`] });
        continue;
      }
      result.sourcesAdded += toInsert.length;
    }

    // Fill summary fields from sourced rows. Only overwrite values still "unknown".
    const update: Record<string, string> = {};
    if (groupRows.some((r) => r.claim_type === "us_ecommerce")) update.us_ecommerce = "verified";
    const cat = groupRows.find((r) => r.claim_type === "category");
    if (cat && company.category === "unknown") update.category = cat.claim;
    const growth = groupRows.find((r) => r.claim_type === "growth_signal");
    if (growth && company.growth_signal === "unknown") update.growth_signal = growth.claim;
    const contact = groupRows.find((r) => r.claim_type === "contact");
    if (contact && company.contact_name === "unknown") {
      const [name, role, email] = contact.claim.split("|").map((p) => p.trim());
      update.contact_name = name;
      update.contact_role = role;
      if (email && !company.contact_email) update.contact_email = email;
    }
    if (Object.keys(update).length > 0) {
      await supabase.from("companies").update(update).eq("id", company.id);
    }
  }

  await supabase.from("audit_logs").insert({
    actor_id: user.id,
    action: "csv_import",
    entity_type: "company_sources",
    detail: {
      rows_received: rows.length,
      companies_created: result.companiesCreated,
      sources_added: result.sourcesAdded,
      duplicates_skipped: result.duplicatesSkipped,
      rows_rejected: result.rejected.length,
    },
  });

  return result;
}

"use client";

import { useState, useTransition } from "react";
import Link from "next/link";
import Papa from "papaparse";
import { importRows, type ImportResult } from "./actions";
import { REQUIRED_COLUMNS, validateRow, type RawRow } from "@/lib/import-schema";

export default function ImportPage() {
  const [rows, setRows] = useState<RawRow[]>([]);
  const [fileName, setFileName] = useState("");
  const [headerError, setHeaderError] = useState<string | null>(null);
  const [result, setResult] = useState<ImportResult | null>(null);
  const [pending, startTransition] = useTransition();

  function onFile(file: File) {
    setResult(null);
    setHeaderError(null);
    setFileName(file.name);
    Papa.parse<RawRow>(file, {
      header: true,
      skipEmptyLines: true,
      transformHeader: (h) => h.trim().toLowerCase(),
      complete: (res) => {
        const missing = REQUIRED_COLUMNS.filter((c) => !(res.meta.fields ?? []).includes(c));
        if (missing.length > 0) {
          setHeaderError(`Missing columns: ${missing.join(", ")}`);
          setRows([]);
          return;
        }
        setRows(res.data);
      },
    });
  }

  const preview = rows.map((r) => validateRow(r));
  const validCount = preview.filter((p) => p.ok).length;
  const invalid = preview
    .map((p, i) => ({ p, line: i + 2 }))
    .filter((x) => !x.p.ok) as { p: { ok: false; errors: string[] }; line: number }[];

  return (
    <div className="max-w-3xl space-y-6">
      <div>
        <Link href="/prospects" className="text-sm text-blue-700">
          ← Prospects
        </Link>
        <h1 className="text-2xl font-semibold mt-2">Import research CSV</h1>
        <p className="text-sm text-gray-600 mt-1">
          One row per sourced fact. Rows missing a source URL, title, date or evidence are
          rejected. Nothing is guessed or filled in for you.
        </p>
      </div>

      <input
        type="file"
        accept=".csv,text/csv"
        onChange={(e) => e.target.files?.[0] && onFile(e.target.files[0])}
        className="block text-sm"
      />

      {headerError && <p className="text-red-600 text-sm">{headerError}</p>}

      {rows.length > 0 && (
        <div className="bg-white border rounded p-4 space-y-3">
          <p className="text-sm">
            <b>{fileName}</b>: {rows.length} rows, {validCount} valid, {invalid.length} will be
            rejected.
          </p>
          {invalid.length > 0 && (
            <ul className="text-sm text-red-700 list-disc pl-5 max-h-48 overflow-auto">
              {invalid.map((x) => (
                <li key={x.line}>
                  Line {x.line}: {x.p.errors.join("; ")}
                </li>
              ))}
            </ul>
          )}
          <button
            disabled={pending || validCount === 0}
            onClick={() => startTransition(async () => setResult(await importRows(rows)))}
            className="bg-blue-600 text-white rounded px-4 py-2 text-sm disabled:opacity-50"
          >
            {pending ? "Importing..." : `Import ${validCount} valid rows`}
          </button>
        </div>
      )}

      {result && (
        <div className="bg-white border rounded p-4 text-sm space-y-2">
          {result.error ? (
            <p className="text-red-600">{result.error}</p>
          ) : (
            <>
              <p>
                Companies created: <b>{result.companiesCreated}</b> · Sources added:{" "}
                <b>{result.sourcesAdded}</b> · Duplicates skipped:{" "}
                <b>{result.duplicatesSkipped}</b> · Rejected: <b>{result.rejected.length}</b>
              </p>
              {result.rejected.length > 0 && (
                <ul className="text-red-700 list-disc pl-5">
                  {result.rejected.map((r, i) => (
                    <li key={i}>
                      {r.line > 0 ? `Line ${r.line}: ` : ""}
                      {r.errors.join("; ")}
                    </li>
                  ))}
                </ul>
              )}
              <Link href="/prospects" className="text-blue-700 underline">
                View prospects
              </Link>
            </>
          )}
        </div>
      )}
    </div>
  );
}

import { CRITERIA, type Criterion } from "./config";

export type SourceRow = {
  id: string;
  claim: string;
  claim_type: string;
  source_url: string;
  source_title: string;
  retrieved_at: string;
  evidence: string;
};

export type CriterionResult = {
  criterion: Criterion;
  met: boolean;
  points: number | null; // null while the weight is not set
  reason: string; // built only from stored claims
  sources: SourceRow[]; // every reason is traceable to these
};

export type ScoreResult = {
  criteria: CriterionResult[];
  disqualifiers: SourceRow[];
  weightsComplete: boolean;
  total: number | null; // null until every weight is set
  maxTotal: number | null;
};

export function scoreCompany(
  sources: SourceRow[],
  criteria: Criterion[] = CRITERIA
): ScoreResult {
  const weightsComplete = criteria.every((c) => c.weight !== null);

  const results: CriterionResult[] = criteria.map((criterion) => {
    const matched = sources.filter((s) => s.claim_type === criterion.claimType);
    const met = matched.length > 0;
    const points =
      criterion.weight === null ? null : met ? criterion.weight : 0;
    const reason = met
      ? matched.map((s) => s.claim).join("; ")
      : "No sourced evidence recorded (unknown)";
    return { criterion, met, points, reason, sources: matched };
  });

  return {
    criteria: results,
    disqualifiers: sources.filter((s) => s.claim_type === "disqualifier"),
    weightsComplete,
    total: weightsComplete
      ? results.reduce((sum, r) => sum + (r.points ?? 0), 0)
      : null,
    maxTotal: weightsComplete
      ? criteria.reduce((sum, c) => sum + (c.weight ?? 0), 0)
      : null,
  };
}

import { test } from "node:test";
import assert from "node:assert/strict";
import { scoreCompany, type SourceRow } from "../lib/scoring/score";
import { CRITERIA } from "../lib/scoring/config";

const src = (claim_type: string, claim: string): SourceRow => ({
  id: `${claim_type}-${claim}`,
  claim,
  claim_type,
  source_url: "https://example.com/x",
  source_title: "Title",
  retrieved_at: "2026-11-12",
  evidence: "Fixture evidence text",
});

const blank = CRITERIA.map((c) => ({ ...c, weight: null }));
const weighted = CRITERIA.map((c) => ({ ...c, weight: 10 }));
const all = [
  src("us_ecommerce", "US shipping"),
  src("contact", "Jane | Founder"),
  src("growth_signal", "Hiring bookkeeper"),
  src("pain_signal", "Multi-channel"),
];

test("blank weights: criteria evaluated, no points, no total", () => {
  const r = scoreCompany(all, blank);
  assert.equal(r.weightsComplete, false);
  assert.equal(r.total, null);
  assert.ok(r.criteria.every((c) => c.met && c.points === null));
});

test("partial weights: still no total", () => {
  const partial = weighted.map((c, i) => (i === 0 ? { ...c, weight: null } : c));
  assert.equal(scoreCompany(all, partial).total, null);
});

test("full weights: total sums points of met criteria", () => {
  const r = scoreCompany([src("us_ecommerce", "US shipping")], weighted);
  assert.equal(r.total, 10);
  assert.equal(r.maxTotal, 40);
});

test("unmet criterion says unknown and carries no sources", () => {
  const r = scoreCompany([], weighted);
  assert.ok(r.criteria.every((c) => !c.met && c.sources.length === 0));
  assert.match(r.criteria[0].reason, /unknown/);
  assert.equal(r.total, 0);
});

test("every met criterion is traceable to at least one source", () => {
  const r = scoreCompany(all, blank);
  assert.ok(r.criteria.filter((c) => c.met).every((c) => c.sources.length > 0));
});

test("disqualifiers are surfaced and never count as criterion evidence", () => {
  const r = scoreCompany([src("disqualifier", "Non-US")], weighted);
  assert.equal(r.disqualifiers.length, 1);
  assert.ok(r.criteria.every((c) => !c.met));
});

export const CLAIM_TYPES = [
  "us_ecommerce",
  "category",
  "growth_signal",
  "pain_signal",
  "disqualifier",
  "contact",
] as const;
export type ClaimType = (typeof CLAIM_TYPES)[number];

export const REQUIRED_COLUMNS = [
  "company_name",
  "website",
  "claim_type",
  "claim",
  "source_url",
  "source_title",
  "retrieved_at",
  "evidence",
] as const;

export type RawRow = Record<string, string>;

export type ValidRow = {
  company_name: string;
  host: string; // normalized hostname, used to deduplicate companies
  claim_type: ClaimType;
  claim: string;
  source_url: string;
  source_title: string;
  retrieved_at: string; // YYYY-MM-DD
  evidence: string;
};

export type RowResult =
  | { ok: true; row: ValidRow }
  | { ok: false; errors: string[] };

export function hostFromUrl(input: string): string | null {
  try {
    const u = new URL(/^https?:\/\//i.test(input) ? input : `https://${input}`);
    if (!u.hostname.includes(".")) return null;
    return u.hostname.replace(/^www\./, "").toLowerCase();
  } catch {
    return null;
  }
}

export function validateRow(raw: RawRow): RowResult {
  const errors: string[] = [];
  const get = (k: string) => (raw[k] ?? "").trim();

  const company_name = get("company_name");
  if (!company_name) errors.push("company_name is missing");

  const host = hostFromUrl(get("website"));
  if (!host) errors.push("website is missing or not a valid URL");

  const claim_type = get("claim_type") as ClaimType;
  if (!CLAIM_TYPES.includes(claim_type))
    errors.push(`claim_type must be one of: ${CLAIM_TYPES.join(", ")}`);

  const claim = get("claim");
  if (!claim) errors.push("claim is missing");

  const source_url = get("source_url");
  if (!/^https?:\/\/\S+\.\S+/i.test(source_url))
    errors.push("source_url must be a full http(s) URL");

  const source_title = get("source_title");
  if (!source_title) errors.push("source_title is missing");

  const retrieved_at = get("retrieved_at");
  const today = new Date().toISOString().slice(0, 10);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(retrieved_at) || isNaN(Date.parse(retrieved_at))) {
    errors.push("retrieved_at must be a real date as YYYY-MM-DD");
  } else if (retrieved_at > today) {
    errors.push("retrieved_at cannot be in the future");
  }

  const evidence = get("evidence");
  if (evidence.length < 10)
    errors.push("evidence must be a quote or summary of at least 10 characters");

  if (claim_type === "contact") {
    const parts = claim.split("|").map((p) => p.trim());
    if (parts.length < 2 || !parts[0] || !parts[1])
      errors.push('contact claim must look like "Name | Role" or "Name | Role | email"');
    if (parts[2] && !/^\S+@\S+\.\S+$/.test(parts[2]))
      errors.push("contact email in claim is not a valid email");
  }

  if (errors.length > 0 || !host) return { ok: false, errors };
  return {
    ok: true,
    row: { company_name, host, claim_type, claim, source_url, source_title, retrieved_at, evidence },
  };
}

"use server";

import { createClient } from "@/lib/supabase/server";
import { validateRow, type RawRow, type ValidRow } from "@/lib/import-schema";

export type ImportResult = {
  error?: string;
  rejected: { line: number; errors: string[] }[];
  companiesCreated: number;
  sourcesAdded: number;
  duplicatesSkipped: number;
};

const empty: ImportResult = {
  rejected: [],
  companiesCreated: 0,
  sourcesAdded: 0,
  duplicatesSkipped: 0,
};

export async function importRows(rows: RawRow[]): Promise<ImportResult> {
  const supabase = await createClient();
  const {
    data: { user },
  } = await supabase.auth.getUser();
  if (!user) return { ...empty, error: "Not signed in." };
  if (rows.length === 0) return { ...empty, error: "The file has no rows." };
  if (rows.length > 2000) return { ...empty, error: "Max 2000 rows per import." };

  // Authoritative validation happens here, on the server.
  const result: ImportResult = { ...empty, rejected: [] };
  const valid: ValidRow[] = [];
  rows.forEach((raw, i) => {
    const r = validateRow(raw);
    if (r.ok) valid.push(r.row);
    else result.rejected.push({ line: i + 2, errors: r.errors }); // +2: header is line 1
  });

  const groups = new Map<string, ValidRow[]>();
  for (const row of valid) {
    groups.set(row.host, [...(groups.get(row.host) ?? []), row]);
  }

  for (const [host, groupRows] of groups) {
    const website = `https://${host}`;

    let { data: company } = await supabase
      .from("companies")
      .select("id, category, growth_signal, contact_name, contact_role, contact_email")
      .eq("website", website)
      .maybeSingle();

    if (!company) {
      const { data: created, error } = await supabase
        .from("companies")
        .insert({ name: groupRows[0].company_name, website })
        .select("id, category, growth_signal, contact_name, contact_role, contact_email")
        .single();
      if (error || !created) {
        result.rejected.push({
          line: 0,
          errors: [`Could not create ${host}: ${error?.message ?? "unknown error"}`],
        });
        continue;
      }
      company = created;
      result.companiesCreated++;
    }

    const { data: existing } = await supabase
      .from("company_sources")
      .select("claim, source_url")
      .eq("company_id", company.id);
    const seen = new Set((existing ?? []).map((s) => `${s.source_url}||${s.claim}`));

    const toInsert = groupRows
      .filter((r) => {
        const key = `${r.source_url}||${r.claim}`;
        if (seen.has(key)) {
          result.duplicatesSkipped++;
          return false;
        }
        seen.add(key);
        return true;
      })
      .map((r) => ({
        company_id: company!.id,
        claim: r.claim,
        claim_type: r.claim_type,
        source_url: r.source_url,
        source_title: r.source_title,
        retrieved_at: r.retrieved_at,
        evidence: r.evidence,
        verified_by: user.id,
      }));

    if (toInsert.length > 0) {
      const { error } = await supabase.from("company_sources").insert(toInsert);
      if (error) {
        result.rejected.push({ line: 0, errors: [`Sources for ${host}: ${error.message}`] });
        continue;
      }
      result.sourcesAdded += toInsert.length;
    }

    // Fill summary fields from sourced rows. Only overwrite values still "unknown".
    const update: Record<string, string> = {};
    if (groupRows.some((r) => r.claim_type === "us_ecommerce")) update.us_ecommerce = "verified";
    const cat = groupRows.find((r) => r.claim_type === "category");
    if (cat && company.category === "unknown") update.category = cat.claim;
    const growth = groupRows.find((r) => r.claim_type === "growth_signal");
    if (growth && company.growth_signal === "unknown") update.growth_signal = growth.claim;
    const contact = groupRows.find((r) => r.claim_type === "contact");
    if (contact && company.contact_name === "unknown") {
      const [name, role, email] = contact.claim.split("|").map((p) => p.trim());
      update.contact_name = name;
      update.contact_role = role;
      if (email && !company.contact_email) update.contact_email = email;
    }
    if (Object.keys(update).length > 0) {
      await supabase.from("companies").update(update).eq("id", company.id);
    }
  }

  await supabase.from("audit_logs").insert({
    actor_id: user.id,
    action: "csv_import",
    entity_type: "company_sources",
    detail: {
      rows_received: rows.length,
      companies_created: result.companiesCreated,
      sources_added: result.sourcesAdded,
      duplicates_skipped: result.duplicatesSkipped,
      rows_rejected: result.rejected.length,
    },
  });

  return result;
}

    setResult(null);
    setHeaderError(null);
    setFileName(file.name);

      header: true,

          setRows([]);

import { CRITERIA, type Criterion } from "./config";

export type SourceRow = {
  id: string;
  claim: string;
  claim_type: string;
  source_url: string;
  source_title: string;
  retrieved_at: string;
  evidence: string;
};

export type CriterionResult = {
  criterion: Criterion;
  met: boolean;
  points: number | null; // null while the weight is not set
  reason: string; // built only from stored claims
  sources: SourceRow[]; // every reason is traceable to these
};

export type ScoreResult = {
  criteria: CriterionResult[];
  disqualifiers: SourceRow[];
  weightsComplete: boolean;
  total: number | null; // null until every weight is set
  maxTotal: number | null;
};

export function scoreCompany(
  sources: SourceRow[],
  criteria: Criterion[] = CRITERIA
): ScoreResult {
  const weightsComplete = criteria.every((c) => c.weight !== null);

  const results: CriterionResult[] = criteria.map((criterion) => {
    const matched = sources.filter((s) => s.claim_type === criterion.claimType);
    const met = matched.length > 0;
    const points =
      criterion.weight === null ? null : met ? criterion.weight : 0;
    const reason = met
      ? matched.map((s) => s.claim).join("; ")
      : "No sourced evidence recorded (unknown)";
    return { criterion, met, points, reason, sources: matched };
  });

  return {
    criteria: results,
    disqualifiers: sources.filter((s) => s.claim_type === "disqualifier"),
    weightsComplete,
    total: weightsComplete
      ? results.reduce((sum, r) => sum + (r.points ?? 0), 0)
      : null,
    maxTotal: weightsComplete
      ? criteria.reduce((sum, c) => sum + (c.weight ?? 0), 0)
      : null,
  };
}

// Single entry point for all LLM calls. Server-side only: never import this
// from a client component, because it reads API keys.

export type AiRequest = {
  system: string;
  prompt: string;
  json?: boolean; // ask the model for JSON only
  temperature?: number;
};

export type AiResponse = { text: string; provider: "gemini" | "groq"; model: string };

export class AiUnavailableError extends Error {
  constructor(public attempts: string[]) {
    super(`No AI provider could answer: ${attempts.join(" | ")}`);
  }
}

type Env = Record<string, string | undefined>;
type FetchLike = typeof fetch;

const TIMEOUT_MS = 30_000;

async function callGemini(
  req: AiRequest, key: string, model: string, f: FetchLike
): Promise<string> {
  const res = await f(
    `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-goog-api-key": key },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: req.system }] },
        contents: [{ role: "user", parts: [{ text: req.prompt }] }],
        generationConfig: {
          temperature: req.temperature ?? 0.2,
          ...(req.json ? { responseMimeType: "application/json" } : {}),
        },
      }),
      signal: AbortSignal.timeout(TIMEOUT_MS),
    }
  );
  if (!res.ok) throw new Error(`gemini HTTP ${res.status}`);
  const data = await res.json();
  const text = (data?.candidates?.[0]?.content?.parts ?? [])
    .map((p: { text?: string }) => p.text ?? "")
    .join("");
  if (!text) throw new Error("gemini returned no text");
  return text;
}

async function callGroq(
  req: AiRequest, key: string, model: string, f: FetchLike
): Promise<string> {
  const res = await f("https://api.groq.com/openai/v1/chat/completions", {
    method: "POST",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${key}` },
    body: JSON.stringify({
      model,
      temperature: req.temperature ?? 0.2,
      messages: [
        { role: "system", content: req.system },
        { role: "user", content: req.prompt },
      ],
      ...(req.json ? { response_format: { type: "json_object" } } : {}),
    }),
    signal: AbortSignal.timeout(TIMEOUT_MS),
  });
  if (!res.ok) throw new Error(`groq HTTP ${res.status}`);
  const data = await res.json();
  const text = data?.choices?.[0]?.message?.content;
  if (!text) throw new Error("groq returned no text");
  return text;
}

export function createAi(env: Env = process.env, f: FetchLike = fetch) {
  return async function generate(req: AiRequest): Promise<AiResponse> {
    const attempts: string[] = [];

    const geminiKey = env.GEMINI_API_KEY;
    const geminiModel = env.GEMINI_MODEL;
    if (geminiKey && geminiModel) {
      try {
        const text = await callGemini(req, geminiKey, geminiModel, f);
        return { text, provider: "gemini", model: geminiModel };
      } catch (e) {
        attempts.push(e instanceof Error ? e.message : "gemini failed");
      }
    } else {
      attempts.push("gemini not configured (need GEMINI_API_KEY and GEMINI_MODEL)");
    }

    const groqKey = env.GROQ_API_KEY;
    const groqModel = env.GROQ_MODEL || "llama-3.3-70b-versatile";
    if (groqKey) {
      try {
        const text = await callGroq(req, groqKey, groqModel, f);
        return { text, provider: "groq", model: groqModel };
      } catch (e) {
        attempts.push(e instanceof Error ? e.message : "groq failed");
      }
    } else {
      attempts.push("groq not configured (need GROQ_API_KEY)");
    }

    throw new AiUnavailableError(attempts);
  };
}

export const generate = createAi();

There are three reasons beyond raw limits:

Prospect messages stay off Gemini. The free tier may use what you send to improve Google products. Public web pages are fine for it. Prospect names and replies should not go there. Groq's data terms are unchecked, so read them before sending anything real. For the case study, the replies are simulated.
Fallback is one-directional on purpose. If Groq fails on a message task, the call fails and the thread escalates to a human. It does not spill to Gemini.
Groq's 8K tokens/min applies to input plus output. gpt-oss models are reasoning models, so thinking tokens count too. Calls must stay small, with no web pages sent to Groq.

Defaults

Gemini: 3.5 Flash Lite, as you chose. Do not use the 20-requests/day models for the pipeline.
Groq: openai/gpt-oss-120b. Reply volume is low, so I'd take quality over speed. I haven't verified it beats the 20b model for this task. We'll run both on the 30-reply test set and put the numbers in HOW_I_WORKED.md.

Budget

Gemini, 500 requests/day. About 140–175 calls per full research run, so you get roughly two full runs a day including testing. The quota resets at midnight Pacific, which is about 09:00 in Algeria in winter. We'll cache fetched pages and extractions so re-runs cost nothing.
Groq, 200K tokens/day. At about 2.5K tokens per call, that's roughly 80 calls a day. Running the 30-reply test through the model alone could eat most of it. So plain code handles the unambiguous cases first (opt-out, price, "are you a bot?"), and the model only sees the ambiguous ones.

// Single entry point for all LLM calls. Server-side only (reads API keys).

export type Task = "extract" | "message";
export type Provider = "gemini" | "groq";

export type AiRequest = {
  task: Task;
  system: string;
  prompt: string;
  json?: boolean;
  temperature?: number;
  maxTokens?: number; // output cap, default 1000
};

export type AiResponse = { text: string; provider: Provider; model: string };

export class AiUnavailableError extends Error {
  constructor(public attempts: string[]) {
    super(`No AI provider could answer: ${attempts.join(" | ")}`);
  }
}

type Env = Record<string, string | undefined>;
type FetchLike = (input: string, init?: RequestInit) => Promise<Response>;
export type Deps = {
  fetch?: FetchLike;
  now?: () => number;
  sleep?: (ms: number) => Promise<void>;
};

const TIMEOUT_MS = 60_000;
const GEMINI_MIN_GAP_MS = 4_500; // stay under 15 requests/min
const GROQ_MIN_GAP_MS = 2_200; // stay under 30 requests/min
const GROQ_TPM_BUDGET = 7_000; // limit is 8K/min, input + output
const GROQ_MAX_CALL_TOKENS = 6_000;

const estimateTokens = (s: string) => Math.ceil(s.length / 3); // deliberately conservative
const errMsg = (e: unknown) => (e instanceof Error ? e.message : "unknown error");

// Prospect messages never go to Gemini's free tier unless explicitly allowed
// (use AI_MESSAGES_ALLOW_GEMINI=1 only with fake test data).
function routeFor(task: Task, env: Env): Provider[] {
  if (task === "extract") return ["gemini", "groq"];
  return env.AI_MESSAGES_ALLOW_GEMINI === "1" ? ["groq", "gemini"] : ["groq"];
}

async function callGemini(req: AiRequest, key: string, model: string, f: FetchLike) {
  const res = await f(
    `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-goog-api-key": key },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: req.system }] },
        contents: [{ role: "user", parts: [{ text: req.prompt }] }],
        generationConfig: {
          temperature: req.temperature ?? 0.2,
          maxOutputTokens: req.maxTokens ?? 1000,
          ...(req.json ? { responseMimeType: "application/json" } : {}),
        },
      }),
      signal: AbortSignal.timeout(TIMEOUT_MS),
    }
  );
  if (!res.ok) throw new Error(`gemini HTTP ${res.status}`);
  const data = await res.json();
  const cand = data?.candidates?.[0];
  if (cand?.finishReason === "MAX_TOKENS") throw new Error("gemini output truncated");
  const text = (cand?.content?.parts ?? [])
    .map((p: { text?: string }) => p.text ?? "")
    .join("");
  if (!text) throw new Error("gemini returned no text");
  return text;
}

async function callGroq(
  req: AiRequest, key: string, model: string, effort: string | undefined, f: FetchLike
) {
  const system = req.json
    ? `${req.system}\n\nReply with a single JSON object and nothing else.`
    : req.system;
  const res = await f("https://api.groq.com/openai/v1/chat/completions", {
    method: "POST",
    headers: { "Content-Type": "application/json", Authorization: `Bearer ${key}` },
    body: JSON.stringify({
      model,
      temperature: req.temperature ?? 0.2,
      max_completion_tokens: req.maxTokens ?? 1000,
      ...(effort ? { reasoning_effort: effort } : {}),
      messages: [
        { role: "system", content: system },
        { role: "user", content: req.prompt },
      ],

  if (choice?.finish_reason === "length") throw new Error("groq output truncated");
  const text = choice?.message?.content;
  if (!text) throw new Error("groq returned no text");
  return text as string;
}

export function createAi(env: Env = process.env, deps: Deps = {}) {
  const f: FetchLike = deps.fetch ?? ((u, i) => fetch(u, i));
  const now = deps.now ?? (() => Date.now());
  const sleep = deps.sleep ?? ((ms: number) => new Promise<void>((r) => setTimeout(r, ms)));

  const lastCall: Record<Provider, number> = { gemini: -Infinity, groq: -Infinity };
  const groqWindow: { t: number; tokens: number }[] = [];

  async function throttle(p: Provider, est: number) {
    const gap = p === "gemini" ? GEMINI_MIN_GAP_MS : GROQ_MIN_GAP_MS;
    for (;;) {
      const t = now();
      let wait = Math.max(0, lastCall[p] + gap - t);
      if (p === "groq") {
        while (groqWindow.length && groqWindow[0].t <= t - 60_000) groqWindow.shift();
        const used = groqWindow.reduce((s, e) => s + e.tokens, 0);
        if (used + est > GROQ_TPM_BUDGET && groqWindow.length) {
          wait = Math.max(wait, groqWindow[0].t + 60_000 - t);
        }
      }
      if (wait <= 0) return;
      await sleep(wait);
    }
  }

  return async function generate(req: AiRequest): Promise<AiResponse> {
    const attempts: string[] = [];

    for (const provider of routeFor(req.task, env)) {
      if (provider === "gemini") {
        const key = env.GEMINI_API_KEY;
        const model = env.GEMINI_MODEL;
        if (!key || !model) {
          attempts.push("gemini not configured (need GEMINI_API_KEY and GEMINI_MODEL)");
          continue;
        }
        try {
          await throttle("gemini", 0);
          const text = await callGemini(req, key, model, f);
          return { text, provider, model };
        } catch (e) {
          attempts.push(errMsg(e));
        } finally {
          lastCall.gemini = now();
        }
      } else {
        const key = env.GROQ_API_KEY;
        const model = env.GROQ_MODEL || "openai/gpt-oss-120b";
        if (!key) {
          attempts.push("groq not configured (need GROQ_API_KEY)");
          continue;
        }
        const est = estimateTokens(req.system + req.prompt) + (req.maxTokens ?? 1000);
        if (est > GROQ_MAX_CALL_TOKENS) {
          attempts.push(`input too large for Groq's tokens-per-minute cap (~${est} tokens)`);
          continue;
        }
        try {
          await throttle("groq", est);
          const text = await callGroq(req, key, model, env.GROQ_REASONING_EFFORT, f);
          return { text, provider, model };
        } catch (e) {
          attempts.push(errMsg(e));
        } finally {
          lastCall.groq = now();
          groqWindow.push({ t: now(), tokens: est });
        }
      }
    }
    throw new AiUnavailableError(attempts);
  };
}

Claude:
Yes, and now is the right time
The version I sent last time has a basic fallback, but only one model per provider. It has three weaknesses:

One model per provider. Your Gemini project has several Flash Lite models, each with its own daily quota in your console. We'd leave most of that unused.
No memory. An exhausted model gets retried on every call, which wastes a request and about 5 seconds each time.
It can't tell "wait a minute" from "done for today". Per-minute limits recover quickly. Per-day limits reset at midnight Pacific time.

Since you haven't applied the earlier version, this one replaces it, so there's no second round of edits.
How it works

A model chain per task, set in .env.local. The code tries models in order, so you can reorder or add models without touching code.
Cooldowns. On a 429 the model is skipped for 60 seconds. A second failure in a row means 10 minutes, and a third means 3 hours. It honors a Retry-After header if the provider sends one. A success resets the count. The code doesn't need to know whether the limit was per-minute or per-day, because the escalation covers both.
Other failures. A server error or timeout means a 30-second pause. A 401, 403 or 404 (bad key or model ID) parks the model for 24 hours and reports it clearly. Truncated or empty output tries the next model with no penalty, because the model isn't at fault.
When every model is cooling down, it throws immediately and doesn't sleep for hours. The caller then shows "AI unavailable" or escalates to a human. It never invents output.
Privacy is enforced in code. For message tasks, any Gemini entry in the chain is dropped unless AI_MESSAGES_ALLOW_GEMINI=1. Typing a Gemini model into the message chain doesn't send prospect messages there.
Groq's 8K tokens/min cap stays enforced per model. Oversized inputs are skipped.

Caveats

Separate quotas are an assumption. Your console lists limits per model, so I assume each model has its own daily pool, giving the Flash Lite models roughly 500 requests/day each. If a second model returns a 429 right after the first, the quota is shared, and we simply rely on one pool.
Quality differs between models. That's safe because our guardrails are code. Quotes must appear word for word in the page, JSON must parse, and prices are blocked. Each result records which model produced it, and we'll store that in the audit log when we wire up the research agent.
Vercel is stateless. Cooldown memory lasts only while a process is warm. On the deployed app, a cold start can cost one wasted failed call, which is acceptable. A batch run through a local script remembers cooldowns for the whole run.
A cap, not a bypass. This uses the quotas you have more fully. It doesn't create new ones.

// Single entry point for all LLM calls. Server-side only (reads API keys).
// Tries a configured chain of models per task, with cooldowns for models
// that are rate-limited or broken.

export type Task = "extract" | "message";
export type Provider = "gemini" | "groq";
export type ModelRef = { provider: Provider; model: string };

export type AiRequest = {
  task: Task;
  system: string;
  prompt: string;
  json?: boolean;
  temperature?: number;
  maxTokens?: number; // output cap, default 1000
};

export type AiResponse = {
  text: string;
  provider: Provider;
  model: string;
  attempts: string[]; // models that failed or were skipped before this one answered
};

export class AiUnavailableError extends Error {
  constructor(public attempts: string[]) {
    super(`No AI model could answer: ${attempts.join(" | ")}`);
  }
}

class AiHttpError extends Error {
  constructor(public status: number, public retryAfterMs?: number) {
    super(`HTTP ${status}`);
  }
}
class AiOutputError extends Error {}

type Env = Record<string, string | undefined>;
type FetchLike = (input: string, init?: RequestInit) => Promise<Response>;
export type Deps = {
  fetch?: FetchLike;
  now?: () => number;
  sleep?: (ms: number) => Promise<void>;
};

const TIMEOUT_MS = 60_000;
const MIN_GAP_MS: Record<Provider, number> = { gemini: 4_500, groq: 2_200 };
const GROQ_TPM_BUDGET = 7_000; // limit is 8K/min, input + output
const GROQ_MAX_CALL_TOKENS = 6_000;
const RATE_LIMIT_COOLDOWNS_MS = [60_000, 600_000, 10_800_000]; // 1 min, 10 min, 3 h
const TRANSIENT_COOLDOWN_MS = 30_000;
const CONFIG_COOLDOWN_MS = 24 * 3_600_000;
const MAX_RETRY_AFTER_MS = 10_800_000;

const estimateTokens = (s: string) => Math.ceil(s.length / 3); // deliberately conservative

function parseRetryAfter(res: Response): number | undefined {
  const v = res.headers.get("retry-after");
  if (!v) return undefined;
  const n = Number(v);
  if (!Number.isFinite(n) || n < 0) return undefined;
  return Math.min(n * 1000, MAX_RETRY_AFTER_MS);
}

function parseChain(raw: string): ModelRef[] {
  return raw
    .split(",")
    .map((s) => s.trim())
    .filter(Boolean)
    .flatMap((entry): ModelRef[] => {
      const i = entry.indexOf(":");
      if (i < 1) return [];
      const provider = entry.slice(0, i);
      const model = entry.slice(i + 1).trim();
      if ((provider !== "gemini" && provider !== "groq") || !model) return [];
      return [{ provider, model }];
    });
}

function chainFor(task: Task, env: Env): ModelRef[] {
  const raw = task === "extract" ? env.AI_EXTRACT_CHAIN : env.AI_MESSAGE_CHAIN;
  const groqDefault = env.GROQ_MODEL || "openai/gpt-oss-120b";
  let chain: ModelRef[];
  if (raw) {
    chain = parseChain(raw);
  } else if (task === "extract") {
    chain = [
      ...(env.GEMINI_MODEL ? [{ provider: "gemini" as const, model: env.GEMINI_MODEL }] : []),
      { provider: "groq" as const, model: groqDefault },
    ];
  } else {
    chain = [{ provider: "groq" as const, model: groqDefault }];
  }
  // Prospect messages never go to Gemini's free tier (it may train on them),
  // unless explicitly allowed. Use the override only with fake test data.
  if (task === "message" && env.AI_MESSAGES_ALLOW_GEMINI !== "1") {
    chain = chain.filter((r) => r.provider !== "gemini");
  }
  return chain;
}

async function callGemini(req: AiRequest, key: string, model: string, f: FetchLike) {
  const res = await f(
    `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json", "x-goog-api-key": key },
      body: JSON.stringify({
        systemInstruction: { parts: [{ text: req.system }] },
        contents: [{ role: "user", parts: [{ text: req.prompt }] }],
        generationConfig: {
          temperature: req.temperature ?? 0.2,
          maxOutputTokens: req.maxTokens ?? 1000,
          ...(req.json ? { responseMimeType: "application/json" } : {}),
        },
      }),
      signal: AbortSignal.timeout(TIMEOUT_MS),
    }
  );
  if (!res.ok) throw new AiHttpError(res.status, parseRetryAfter(res));

    .join("");

function describe(e: unknown): string {
  if (e instanceof AiHttpError) {
    const hint = [401, 403, 404].includes(e.status) ? " (check the API key and model ID)" : "";
    return `${e.message}${hint}`;
  }
  return e instanceof Error ? e.message : "unknown error";
}

export function createAi(env: Env = process.env, deps: Deps = {}) {
  const f: FetchLike = deps.fetch ?? ((u, i) => fetch(u, i));
  const now = deps.now ?? (() => Date.now());
  const sleep = deps.sleep ?? ((ms: number) => new Promise<void>((r) => setTimeout(r, ms)));

  const cooldownUntil = new Map<string, number>();
  const strikes = new Map<string, number>();
  const lastCall: Record<Provider, number> = { gemini: -Infinity, groq: -Infinity };
  const groqWindows = new Map<string, { t: number; tokens: number }[]>();
  const idOf = (r: ModelRef) => `${r.provider}:${r.model}`;

  async function throttle(ref: ModelRef, est: number) {
    for (;;) {
      const t = now();
      let wait = Math.max(0, lastCall[ref.provider] + MIN_GAP_MS[ref.provider] - t);
      if (ref.provider === "groq") {
        const w = groqWindows.get(idOf(ref)) ?? [];
        while (w.length && w[0].t <= t - 60_000) w.shift();
        groqWindows.set(idOf(ref), w);
        const used = w.reduce((s, e) => s + e.tokens, 0);
        if (used + est > GROQ_TPM_BUDGET && w.length) {
          wait = Math.max(wait, w[0].t + 60_000 - t);
        }
      }
      if (wait <= 0) return;
      await sleep(wait);
    }
  }

  function penalize(id: string, e: unknown) {
    const t = now();
    if (e instanceof AiOutputError) return; // the model isn't at fault
    if (e instanceof AiHttpError) {
      if (e.status === 429) {
        const n = (strikes.get(id) ?? 0) + 1;
        strikes.set(id, n);
        const ms =
          e.retryAfterMs ??
          RATE_LIMIT_COOLDOWNS_MS[Math.min(n - 1, RATE_LIMIT_COOLDOWNS_MS.length - 1)];
        cooldownUntil.set(id, t + ms);
      } else if ([401, 403, 404].includes(e.status)) {
        cooldownUntil.set(id, t + CONFIG_COOLDOWN_MS);
      } else if (e.status >= 500) {
        cooldownUntil.set(id, t + TRANSIENT_COOLDOWN_MS);
      }
      return; // other 4xx (such as 400) may be input-specific: no cooldown
    }
    cooldownUntil.set(id, t + TRANSIENT_COOLDOWN_MS); // network error or timeout
  }

  async function generate(req: AiRequest): Promise<AiResponse> {
    const attempts: string[] = [];
    const chain = chainFor(req.task, env);
    if (chain.length === 0) {
      throw new AiUnavailableError([`no models configured for task "${req.task}"`]);
    }

    for (const ref of chain) {
      const id = idOf(ref);
      const t0 = now();
      const until = cooldownUntil.get(id) ?? 0;
      if (t0 < until) {
        attempts.push(`${id}: cooling down for ${Math.ceil((until - t0) / 1000)}s`);
        continue;
      }

      const key = ref.provider === "gemini" ? env.GEMINI_API_KEY : env.GROQ_API_KEY;
      if (!key) {
        attempts.push(`${id}: no API key configured`);
        continue;
      }

      const est = estimateTokens(req.system + req.prompt) + (req.maxTokens ?? 1000);
      if (ref.provider === "groq" && est > GROQ_MAX_CALL_TOKENS) {
        attempts.push(`${id}: input too large for the tokens-per-minute cap (~${est} tokens)`);
        continue;
      }

      try {
        await throttle(ref, est);
        const text =
          ref.provider === "gemini"
            ? await callGemini(req, key, ref.model, f)
            : await callGroq(req, key, ref.model, env.GROQ_REASONING_EFFORT, f);
        strikes.delete(id);
        return { text, provider: ref.provider, model: ref.model, attempts };
      } catch (e) {
        attempts.push(`${id}: ${describe(e)}`);
        penalize(id, e);
      } finally {
        lastCall[ref.provider] = now();
        if (ref.provider === "groq") {
          const w = groqWindows.get(id) ?? [];
          w.push({ t: now(), tokens: est });
          groqWindows.set(id, w);
        }
      }
    }
    throw new AiUnavailableError(attempts);
  }

  // Which models are currently cooling down (useful for a status panel later).
  function status() {
    const t = now();
    return [...cooldownUntil]
      .filter(([, until]) => until > t)
      .map(([model, until]) => ({ model, availableAt: new Date(until).toISOString() }));
  }

  return { generate, status };
}

export const { generate, status: aiStatus } = createAi();

I built this as a local script that writes a CSV, not as a button in the app. That choice is deliberate:

A run takes minutes, because of rate limits and page fetches, and Vercel would time out.
The CSV goes through the same import page you already tested, so the source rules stay in one place.
Reviewing the CSV is the human verification step, which is what the case wants.

Each company goes through five steps:

Jina Reader fetches the homepage. The script picks up to 3 same-site links (about, shipping, careers, press and similar), plus any extra pages you supply, such as a press article about a funding round.
Gemini reads those pages and proposes claims with quoted evidence.
Code checks that every quote appears word for word in a page we actually fetched. The source URL must be a fetched page. The claim type must be one of the seven. For contacts, the name and role must both appear in the quote, and any email must appear on the page. Anything that fails is dropped, with the reason logged.
The script sets the retrieval date and the page title, never the model.
Survivors pass through the importer's own validation, then go into the CSV, with a report of what was dropped and why.

Page text is treated as untrusted data, so a page that says "ignore your instructions" can't steer the output. The verbatim check is the backstop for that.
Fetched pages and extractions are cached in .cache/, so a re-run costs no extra requests. A run uses about one Gemini request per company, which is small against your 500/day.

export type Page = { url: string; title: string; text: string };
export type InputCompany = { name: string; website: string; extras: string[] };

// Strip images, turn markdown links into their text, tidy whitespace.
export function cleanPage(raw: string): string {
  return raw
    .replace(/!\[[^\]]*\]\([^)]*\)/g, "")
    .replace(/\[([^\]]*)\]\([^)]*\)/g, "$1")
    .replace(/[ \t]+/g, " ")
    .replace(/\n{3,}/g, "\n\n")
    .trim();
}

export function parseJinaTitle(raw: string, fallback: string): string {
  const m = raw.match(/^Title:\s*(.+)$/m);
  return (m?.[1] ?? "").trim() || fallback;
}

export function hostOf(url: string): string | null {
  try {
    return new URL(url).hostname.replace(/^www\./, "").toLowerCase();
  } catch {
    return null;
  }
}

const PRIORITY = [
  "about", "our-story", "story", "shipping", "careers", "jobs",
  "press", "news", "wholesale", "stockist", "team", "faq", "contact",
];
const SKIP_EXT = /\.(jpe?g|png|gif|webp|svg|pdf|zip|css|js|ico|mp4)$/i;

// Pick same-site pages likely to hold ICP evidence, best first.
export function pickLinks(markdown: string, baseUrl: string, max = 3): string[] {
  const baseHost = hostOf(baseUrl);
  const home = baseUrl.replace(/\/+$/, "");
  const found = new Map<string, number>();
  for (const m of markdown.matchAll(/\]\(([^)\s]+)/g)) {
    let u: URL;
    try {
      u = new URL(m[1], baseUrl);
    } catch {
      continue;
    }
    if (u.protocol !== "http:" && u.protocol !== "https:") continue;
    if (hostOf(u.href) !== baseHost) continue;
    if (SKIP_EXT.test(u.pathname)) continue;
    u.hash = "";
    u.search = "";
    const href = u.href.replace(/\/+$/, "");
    if (href === home) continue;
    const path = u.pathname.toLowerCase();
    const rank = PRIORITY.findIndex((k) => path.includes(k));
    if (rank < 0) continue;
    if (!found.has(href) || rank < found.get(href)!) found.set(href, rank);
  }
  return [...found].sort((a, b) => a[1] - b[1]).slice(0, max).map(([h]) => h);
}

// Input format, one company per line:
//   Company name | https://homepage | optional extra page | optional extra page
export function parseInput(text: string): { companies: InputCompany[]; errors: string[] } {
  const companies: InputCompany[] = [];
  const errors: string[] = [];
  const httpOk = (u: string) => /^https?:\/\/\S+\.\S+$/i.test(u);
  text.split("\n").forEach((raw, i) => {
    const line = raw.trim();
    if (!line || line.startsWith("#")) return;
    const [name, website, ...rest] = line.split("|").map((s) => s.trim());
    const extras = rest.filter(Boolean);
    if (!name) errors.push(`Line ${i + 1}: missing company name`);
    else if (!website || !httpOk(website))
      errors.push(`Line ${i + 1}: homepage must be a full http(s) URL`);
    else if (extras.some((e) => !httpOk(e)))
      errors.push(`Line ${i + 1}: extra pages must be full http(s) URLs`);
    else companies.push({ name, website: website.replace(/\/+$/, ""), extras });
  });
  return { companies, errors };
}

import { CLAIM_TYPES, type ClaimType } from "../import-schema";
import type { Page } from "./pages";

export type Proposed = {
  claim_type?: unknown;
  claim?: unknown;
  source_url?: unknown;
  evidence?: unknown;
};
export type AcceptedClaim = {
  claim_type: ClaimType;
  claim: string;
  source_url: string;
  source_title: string;
  evidence: string;
};
export type RejectedClaim = { claim: string; reason: string };

export const MIN_QUOTE = 20;
export const MAX_QUOTE = 300;
export const MAX_PER_TYPE = 3;

// Makes quote matching robust to case, whitespace, curly quotes and markdown noise.
export function normalize(s: string): string {
  return s
    .replace(/[\u2018\u2019\u201B]/g, "'")
    .replace(/[\u201C\u201D]/g, '"')
    .replace(/[\u2013\u2014]/g, "-")
    .replace(/\u00a0/g, " ")
    .replace(/[*_`#>]/g, "")
    .replace(/\s+/g, " ")
    .trim()
    .toLowerCase();
}

const urlKey = (u: string) => u.trim().replace(/#.*$/, "").replace(/\/+$/, "").toLowerCase();

// The core guardrail: the model proposes, this code decides. A claim survives
// only if its evidence is copied verbatim from a page we actually fetched.
export function verifyClaims(proposed: Proposed[], pages: Page[]) {
  const byUrl = new Map(pages.map((p) => [urlKey(p.url), p]));
  const text = new Map(pages.map((p) => [urlKey(p.url), normalize(p.text)]));
  const accepted: AcceptedClaim[] = [];
  const rejected: RejectedClaim[] = [];
  const perType = new Map<string, number>();
  const seen = new Set<string>();

  for (const p of proposed) {
    const claim = typeof p.claim === "string" ? p.claim.trim() : "";
    const reject = (reason: string) => rejected.push({ claim: claim || "(empty)", reason });

    const type = p.claim_type as ClaimType;
    if (!CLAIM_TYPES.includes(type)) { reject("unknown claim_type"); continue; }
    if (!claim || claim.length > 300) { reject("claim empty or too long"); continue; }

    const key = typeof p.source_url === "string" ? urlKey(p.source_url) : "";
    const page = byUrl.get(key);
    if (!page) { reject("source_url is not a page we fetched"); continue; }

    const evidence = typeof p.evidence === "string" ? p.evidence.trim() : "";
    if (evidence.length < MIN_QUOTE || evidence.length > MAX_QUOTE) {
      reject(`evidence must be ${MIN_QUOTE}-${MAX_QUOTE} characters`);
      continue;
    }
    const pageText = text.get(key)!;
    const quote = normalize(evidence);
    if (!pageText.includes(quote)) { reject("evidence not found verbatim on the page"); continue; }

    if (type === "contact") {
      const [name, role, email] = claim.split("|").map((s) => s.trim());
      if (!name || !role) { reject('contact must look like "Name | Role"'); continue; }
      if (!quote.includes(normalize(name)) || !quote.includes(normalize(role))) {
        reject("contact name and role must both appear in the quote");
        continue;
      }
      if (email && !pageText.includes(normalize(email))) {
        reject("contact email does not appear on the page");
        continue;
      }
    }

    const dupKey = `${type}||${claim.toLowerCase()}`;
    if (seen.has(dupKey)) { reject("duplicate claim"); continue; }
    if ((perType.get(type) ?? 0) >= MAX_PER_TYPE) {
      reject(`more than ${MAX_PER_TYPE} claims of type ${type}`);
      continue;
    }
    seen.add(dupKey);
    perType.set(type, (perType.get(type) ?? 0) + 1);
    accepted.push({
      claim_type: type,
      claim,
      source_url: page.url, // canonical URL we fetched, never the model's spelling
      source_title: page.title,
      evidence,
    });
  }
  return { accepted, rejected };
}

      if (got) pages.push(got.page);
    }
    report.push(`- Pages read: ${pages.map((p) => p.url).join(", ")}`);

    const cacheFile = `.cache/extract/${sha(c.website + "|" + pages.map((p) => p.url).join("|"))}.json`;
    let proposed: Proposed[];
    let model = "cache";
    if (fs.existsSync(cacheFile)) {
      proposed = JSON.parse(fs.readFileSync(cacheFile, "utf8")).claims;
    } else {
      const prompt =
        `Company: ${c.name}\n\n` +
        pages.map((p) => `=== PAGE ${p.url} ===\n${p.text}`).join("\n\n");
      try {
        const r = await generate({ task: "extract", system: SYSTEM, prompt, json: true, maxTokens: 3000 });
        model = `${r.provider}:${r.model}`;
        const parsed = extractJson<{ claims?: unknown }>(r.text);
        proposed = Array.isArray(parsed.claims) ? (parsed.claims as Proposed[]) : [];
        fs.writeFileSync(cacheFile, JSON.stringify({ model, claims: proposed }));
      } catch (e) {
        if (e instanceof AiUnavailableError) {
          stoppedAt = c.name;
          report.push(`- AI unavailable: ${e.attempts.join(" | ")}`, "- Run stopped here.", "");
          break;
        }
        report.push(`- Model reply could not be used (${e instanceof Error ? e.message : "error"}). Skipped.`, "");
        continue;
      }
    }

    const { accepted, rejected } = verifyClaims(proposed, pages);
    let kept = 0;
    for (const a of accepted) {
      const row = {
        company_name: c.name,
        website: c.website,
        claim_type: a.claim_type,
        claim: a.claim,
        source_url: a.source_url,
        source_title: a.source_title,
        retrieved_at: today,
        evidence: a.evidence,
      };
      const v = validateRow(row);
      if (v.ok) {
        rows.push(row);
        kept++;
      } else {
        rejected.push({ claim: a.claim, reason: `importer rule: ${v.errors.join("; ")}` });
      }
    }
    report.push(`- Model: ${model}. Proposed ${proposed.length}, kept ${kept}, dropped ${rejected.length}.`);
    for (const x of rejected) report.push(`  - dropped "${x.claim}": ${x.reason}`);
    report.push(...notes.map((n) => `- note: ${n}`), "");
  }

  const stamp = new Date().toISOString().slice(0, 16).replace(/[:T]/g, "-");
  const csvPath = `docs/research-output/research-${stamp}.csv`;
  const reportPath = `docs/research-output/research-${stamp}-report.md`;
  const fields = [...REQUIRED_COLUMNS];
  fs.writeFileSync(csvPath, Papa.unparse({ fields, data: rows.map((r) => fields.map((f) => r[f])) }) + "\n");
  fs.writeFileSync(reportPath, report.join("\n") + "\n");

  console.log(`\nWrote ${rows.length} rows to ${csvPath}`);
  console.log(`Report: ${reportPath}`);
  if (stoppedAt) console.log(`Stopped early at "${stoppedAt}". Re-run later: cached pages are not fetched again.`);
  console.log("NEXT: review every row against its source before importing.");
}

Claude:
claim vs claim_type
Each CSV row records one fact about a company. The two columns describe that fact in different ways.
claim is the fact itself, in plain words. It's what a salesperson reads on the company page and what an email draft can reuse, for example "Ships bikes to US customers (48 contiguous states and Canada)".
claim_type is a label from a fixed list of seven. It tells the app which scoring criterion this fact supports. The app can't read your sentence and guess its meaning, so the label is how the fact reaches the right slot on the company page.

The claim_type decides where the fact shows up. A growth_signal row appears under "Growth signal" on the company page and counts toward that criterion. A disqualifier row triggers the red box. evidence is the proof for the claim, a short quote or summary from the page. claim says what is true and evidence shows where the page says so. source_url is the page the evidence came from.
A quick test for each row: if you cover the claim, could a stranger read the evidence on that source_url and arrive at the same statement? If not, the row needs fixing.
